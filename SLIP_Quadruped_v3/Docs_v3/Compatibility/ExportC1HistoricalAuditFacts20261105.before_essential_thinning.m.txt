function catalog=ExportC1HistoricalAuditFacts20261105(rosterReference,outputFile,options)
%EXPORTC1HISTORICALAUDITFACTS20261105 One bounded historical-facts store.
% Structural retention only: no Root, model, replay, classifier or deletion.
% Actual native orbit cores and final small matrices are collected before
% known execution tapes are omitted. Unknown objects/ambiguous large values
% hold their original file. The aggregate is the only mutable checkpoint;
% interrupted pure-data projection may repeat at most one eight-file chunk.
    entered=tic;u0=unixNow();o=defaults(options);
    limit=struct('entered',entered,'unix_start',u0,'seconds',o.MaxWallSeconds,'reserve',o.SaveMarginSeconds);
    rosterReference=checked(rosterReference);roster=jsondecode(fileread(rosterReference.file));
    if strcmp(roster.schema,'C1-immediate-non-native-audit-retirement-classification-v1')
        assert(numel(roster.hold_native_or_binary)==464,'AuditFacts:Roster','Keep the original464-source roster complete.');sources=cells(roster.hold_native_or_binary);
    else
        assert(strcmp(roster.schema,'C1-explicit-native-facts-source-roster-v1'),'AuditFacts:Roster','Use a closed explicit native-facts roster.');sources=cells(roster.source_references);
    end
    N=numel(sources);assert(N>=1&&N<=1024,'AuditFacts:Roster','Use1..1024 explicit sources.');
    paths=cellfun(@(r)canonical(r.file),sources,'UniformOutput',false);
    assert(numel(unique(paths))==N,'AuditFacts:Roster','Every source occurs once.');
    outputFile=canonical(outputFile);assert(isfolder(fileparts(outputFile))&&~any(strcmp(outputFile,paths)), ...
        'AuditFacts:Output','Use an existing destination directory and a separate aggregate.');
    names={'ExportC1HistoricalAuditFacts20261105','CompactResearchBranch_v3','RecordedReturnPolicy_v3', ...
        'OrbitContentIdentity_v3','C1ScientificDigest_v3','RoundSHA256_v3'};
    methods=cell(numel(names),1);for k=1:numel(names),methods{k}=reference(which(names{k}));end
    configuration=struct('schema','C1-historical-audit-facts-codec-v1','roster_reference',rosterReference, ...
        'methods',{methods},'small_numeric_elements',4096,'primary_table_elements',1000000, ...
        'maximum_source_bytes',256*1024^2,'maximum_visits',500000,'maximum_primary_records',10000, ...
        'maximum_bytes',o.MaximumBytes,'checkpoint_files',8);
    digest=C1ScientificDigest_v3(configuration);priorSHA='';
    if isempty(fieldnames(o.ResumeReference))
        assert(~isfile(outputFile)&&~isfile([outputFile,'.pending.mat'])&&~isfile([outputFile,'.json']), ...
            'AuditFacts:Fresh','Use fresh output paths or an exact explicit resume reference.');
        rows=cell(N,1);for k=1:N,rows{k}=struct('inventory_reference',sources{k}, ...
            'source_reference',struct(),'status','NOT_ASSESSED','retirement_eligible',false, ...
            'primary_records',{{}},'selected_final_facts',{{}},'facts',struct(), ...
            'omissions',{{}},'unknowns',{{}},'variables',struct([]),'source_posthash',false,'failure',struct());end
        catalog=struct('schema','C1-historical-essential-audit-facts-v1','configuration',configuration, ...
            'configuration_digest',digest,'rows',{rows},'cursor',1,'completed',false,'source_posthash',false, ...
            'invocations',{{}},'new_scientific_services',0,'original_sources_removed',0, ...
            'scope',['Historical literal cores, decisions, configurations and final matrices only. ', ...
            'Dense trajectories, failed/FD/iteration/evaluation histories and runtime graphs are unavailable. ', ...
            'No original experiment is reproduced, no pending outcome reconciled, no admission upgraded. ', ...
            'A retirement-eligible row still requires the external owner last-consumer/process review.']);
    else
        resume=checked(o.ResumeReference);assert(strcmp(resume.file,outputFile),'AuditFacts:Resume','Resume the same owned aggregate.');
        z=load(outputFile,'catalog');catalog=z.catalog;clear z
        assert(strcmp(catalog.schema,'C1-historical-essential-audit-facts-v1') ...
            &&strcmp(C1ScientificDigest_v3(catalog.configuration),catalog.configuration_digest), ...
            'AuditFacts:Epoch','Require the intact exact aggregate.');
        if ~isequaln(catalog.configuration,configuration)
            assert(o.AdoptReviewedCodec&&isequaln(rmfield(catalog.configuration,'methods'),rmfield(configuration,'methods')), ...
                'AuditFacts:Epoch','A reviewed codec successor may not change the roster or retention bounds.');
            oldMethods=catalog.configuration.methods;
            assert(numel(oldMethods)==numel(methods),'AuditFacts:Epoch','Provider roster changed.');
            for mi=1:numel(methods)
                assert(strcmp(oldMethods{mi}.file,methods{mi}.file),'AuditFacts:Epoch','Provider location changed.');
                if mi>1,assert(isequaln(oldMethods{mi},methods{mi}),'AuditFacts:Epoch','Only this reviewed exporter codec may change.');end
            end
            oldConfiguration=catalog.configuration;oldDigest=catalog.configuration_digest;
            catalog.configuration=configuration;catalog.configuration_digest=digest;
            if ~isfield(catalog,'codec_successors'),catalog.codec_successors={};end
            catalog.codec_successors{end+1,1}=struct('previous_reference',resume,'previous_configuration',oldConfiguration, ...
                'previous_configuration_digest',oldDigest,'new_configuration_digest',digest,'successful_sources_reread',0);
        end
        if ~isempty(o.RetryHeldIndices)
            assert(o.AdoptReviewedCodec,'AuditFacts:Retry','Explicitly declare reviewed held-source retry.');
            assert(all(o.RetryHeldIndices(:)<=N),'AuditFacts:Retry','Held-source index exceeds this exact roster.');
            previous={};
            for ri=o.RetryHeldIndices(:).'
                old=catalog.rows{ri};assert(~old.retirement_eligible&&startsWith(old.status,'RAW_HELD'), ...
                    'AuditFacts:Retry','Never reset successful or protected authority rows.');previous{end+1,1}=struct('source_index',ri,'literal_previous_outcome',old);
                old.status='NOT_ASSESSED_REVIEWED_SUCCESSOR';catalog.rows{ri}=old;
            end
            if ~isfield(catalog,'explicit_held_retries'),catalog.explicit_held_retries={};end
            catalog.explicit_held_retries{end+1,1}=struct('previous_reference',resume,'rows',{previous},'prepared_utc',stamp());
            catalog.cursor=min(catalog.cursor,min(o.RetryHeldIndices));catalog.completed=false;
        end
        priorSHA=resume.sha256;
    end
    catalog.source_posthash=false;catalog.invocations{end+1,1}=struct('started_utc',stamp(), ...
        'maximum_seconds',o.MaxWallSeconds,'maximum_new_files',o.MaxNewFiles,'resume_reference',o.ResumeReference);
    persist();processed=0;sinceSave=0;
    while catalog.cursor<=N&&processed<o.MaxNewFiles
        if remaining(limit)<=o.SaveMarginSeconds+30,break;end
        at=catalog.cursor;item=sources{at};row=catalog.rows{at};clock=tic;
        if ~startsWith(row.status,'NOT_ASSESSED'),catalog.cursor=at+1;continue;end
        row.failure=struct(); % The literal old held outcome stays in retry history.
        try
            source=seal(item);
            if isfield(row.source_reference,'sha256'),assert(strcmp(row.source_reference.sha256,source.sha256), ...
                    'AuditFacts:RetrySource','A previously observed held source changed generation.');end
            row.source_reference=source;
            vars=whos('-file',source.file);row.variables=vars;
            if source.bytes>configuration.maximum_source_bytes
                row.status='RAW_HELD_SOURCE_SIZE';row.unknowns={struct('path','source','reason','source exceeds bounded single-file intake')};
            else
                state=struct('source',source,'primary',{{}},'primary_digests',{{}},'selected',{{}}, ...
                    'omissions',{{}},'unknowns',{{}},'visits',0,'trajectories',0,'owned_trajectories',{{}},'ambiguous_trajectories',{{}},'limit',limit,'rules',configuration);
                wholeFinite=endsWith(source.file,[filesep,'PK30h_20261102_Current_Finite13_Resolution02.mat']);
                if wholeFinite
                    data=load(source.file,'-mat');C1ScientificDigest_v3(data);row.facts=data;
                    row.status='KEEP_ORIGINAL_COMPACT_FINITE13_AUTHORITY';
                    row.unknowns={struct('path','source','reason','deliberately protected finite13 authority; original remains retained')};
                else
                    context=struct();contextNames=intersect({vars.name},{'report','raw_report','native_report','compacted_report'});
                    if ~isempty(contextNames)
                        q=load(source.file,'-mat',contextNames{:});
                        for ci=1:numel(contextNames),v=q.(contextNames{ci});if isstruct(v)&&isscalar(v),context=v;break;end;end
                        clear q v
                    end
                    facts=struct();
                    for vi=1:numel(vars)
                        check(limit,false);name=vars(vi).name;data=load(source.file,'-mat',name);value=data.(name);clear data
                        state=scout(value,name,context,state,0);
                        [facts.(name),state]=project(value,name,state,0);clear value
                    end
                    clear context
                    if ~isempty(state.ambiguous_trajectories)
                        state=unknown(state,'source',['trajectory without a complete primary context: ',strjoin(state.ambiguous_trajectories(1:min(8,end)),'; ')]);
                    end
                    row.primary_records=state.primary;row.selected_final_facts=state.selected;
                    row.omissions=state.omissions;row.unknowns=state.unknowns;
                    if isempty(state.unknowns)
                        C1ScientificDigest_v3(facts);row.facts=facts;
                        row.status='ESSENTIAL_FACTS_PROJECTED';row.retirement_eligible=true;
                    else
                        row.status='RAW_HELD_UNASSESSED';row.facts=struct('projection_not_authorized',true);
                        % Partial observations cannot replace a held original.
                        row.primary_records={};row.selected_final_facts={};
                    end
                    clear facts state
                end
            end
            checked(source);row.source_posthash=true;
        catch problem
            row.status='RAW_HELD_NATIVE_EXCEPTION';row.retirement_eligible=false;
            row.facts=struct();row.primary_records={};row.selected_final_facts={};
            row.failure=struct('identifier',problem.identifier,'message',problem.message,'stack',problem.stack);
        end
        row.elapsed_seconds=toc(clock);catalog.rows{at}=row;catalog.cursor=at+1;
        processed=processed+1;sinceSave=sinceSave+1;
        if sinceSave>=8,persist();sinceSave=0;end
        fprintf('HISTORICAL_AUDIT_FACT source=%d/%d status=%s primary=%d seconds=%.3f\n', ...
            at,N,row.status,numel(row.primary_records),row.elapsed_seconds);
    end
    for k=1:numel(methods),checked(methods{k});end;checked(rosterReference);
    catalog.completed=catalog.cursor==N+1;catalog.source_posthash=true;
    catalog.invocations{end}.finished_utc=stamp();catalog.invocations{end}.processed=processed;
    catalog.retirement_eligible_count=sum(cellfun(@(r)r.retirement_eligible,catalog.rows));
    catalog.raw_held_count=sum(cellfun(@(r)startsWith(r.status,'RAW_HELD')||startsWith(r.status,'KEEP_ORIGINAL'),catalog.rows));
    catalog.last_elapsed_seconds=toc(entered);persist();
    fprintf('HISTORICAL_AUDIT_FACTS_RETURN cursor=%d complete=%d eligible=%d held=%d\n', ...
        catalog.cursor,catalog.completed,catalog.retirement_eligible_count,catalog.raw_held_count);

    function persist()
        check(limit,true);
        if isempty(priorSHA),assert(~isfile(outputFile),'AuditFacts:CAS','A concurrent aggregate appeared.');
        else,assert(strcmp(RoundSHA256_v3(outputFile),priorSHA),'AuditFacts:CAS','The owned aggregate changed.');end
        stage=[outputFile,'.pending.mat'];assert(~isfile(stage),'AuditFacts:Stage','An unknown owned stage is held.');
        save(stage,'catalog','-v7');a=load(stage,'catalog');b=load(stage,'catalog');
        assert(isequaln(a.catalog,catalog)&&isequaln(b.catalog,catalog),'AuditFacts:Read2','The plain aggregate changed in native read2.');check(limit,true);
        d=dir(stage);assert(d.bytes<=o.MaximumBytes,'AuditFacts:Size','Keep originals and unqualified stage: aggregate exceeds its explicit cap.');
        [ok,msg]=movefile(stage,outputFile,'f');assert(ok,'AuditFacts:Publish','%s',msg);priorSHA=RoundSHA256_v3(outputFile);check(limit,true);
        index=rmfield(catalog,'rows');index.rows=cellfun(@(r)struct('inventory_reference',r.inventory_reference, ...
            'source_reference',r.source_reference,'status',r.status,'retirement_eligible',r.retirement_eligible, ...
            'primary_count',numel(r.primary_records),'omission_count',numel(r.omissions), ...
            'unknowns',{r.unknowns},'source_posthash',r.source_posthash,'failure',r.failure),catalog.rows,'UniformOutput',false);
        text=jsonencode(index);encoded=unicode2native(text,'UTF-8');stage=[outputFile,'.json.pending'];assert(~isfile(stage),'AuditFacts:Stage','An unknown JSON stage is held.');
        fid=fopen(stage,'w');assert(fid>=0,'AuditFacts:Open','Cannot create index stage.');closer=onCleanup(@()fclose(fid));
        assert(fwrite(fid,encoded,'uint8')==numel(encoded),'AuditFacts:Write','Short JSON index write.');clear closer
        [ok,msg]=movefile(stage,[outputFile,'.json'],'f');assert(ok,'AuditFacts:Publish','%s',msg);
    end
end

function state=scout(v,path,context,state,depth)
    state=visit(state,depth,path);cls=class(v);
    if isa(v,'HybridOrbit_v3')
        for k=1:numel(v)
            o=v(k);core=struct('raw_state14',o.initial_state,'mode',o.initial_mode,'parameter',o.parameter, ...
                'period',o.period,'stride_displacement',o.stride_displacement,'poincare_state',o.poincare_state, ...
                'return_policy',o.return_policy,'return_policy_name',o.return_policy_name,'return_multiplicity',o.return_multiplicity, ...
                'section_chart',o.section_chart,'event_history',o.event_history,'event_clusters',o.event_clusters);
            [~,policy]=RecordedReturnPolicy_v3(context);core.recorded_policy_evidence=policy;
            values=struct();for name=properties(o).',values.(name{1})=o.(name{1});end
            state.owned_trajectories{end+1,1}=[path,'.public_properties(1).trajectory'];
            % All public properties are inspected. This catches an unknown
            % object hidden in metadata rather than silently pruning it.
            state=scout(values,[path,'.public_properties'],context,state,depth+1);
            [kept,state]=project(values,[path,'.public_properties'],state,depth+1);
            [core,state]=project(core,[path,'.literal_primary'],state,depth+1);
            completeCore=numel(o.initial_state)==14&&numel(o.initial_mode)==4&&numel(o.parameter)==10 ...
                &&isscalar(o.period)&&numel(o.stride_displacement)==14;
            raw29=[];if completeCore,raw29=[o.initial_state(:);o.stride_displacement(:);o.period];end
            extra=struct('floquet_multiplier',o.floquet_multiplier,'stability',o.stability, ...
                'primitive_cycle',o.primitive_cycle,'gait_identification',o.gait_identification);
            [extra,state]=project(extra,[path,'.primary_diagnostics'],state,depth+1);
            nativeID='';identityFailure=struct();try,nativeID=OrbitContentIdentity_v3(o);catch p,identityFailure=struct('identifier',p.identifier,'message',p.message);end
            record=struct('source_paths',{{path}},'native_id',nativeID,'native_id_failure',identityFailure, ...
                'source_array_index',k,'source_container_size',size(v), ...
                'complete_core29_available',completeCore,'core29',raw29, ...
                'core',core,'saved_report_status',reportHead(context),'diagnostics',extra,'public_properties',kept);
            key=C1ScientificDigest_v3(rmfield(record,'source_paths'));hit=find(strcmp(state.primary_digests,key),1);
            if isempty(hit)
                assert(numel(state.primary)<state.rules.maximum_primary_records,'AuditFacts:PrimaryCap','Too many primary records; keep raw.');
                state.primary{end+1,1}=record;state.primary_digests{end+1,1}=key;
            elseif numel(state.primary{hit}.source_paths)<256
                state.primary{hit}.source_paths{end+1,1}=path;
            end
        end
    elseif isa(v,'Trajectory_v3')
        state.trajectories=state.trajectories+1;
        if ~knownFailedContext(context)&&~any(strcmp(state.owned_trajectories,path))
            state.ambiguous_trajectories{end+1,1}=path;
        elseif knownFailedContext(context)
            state=omit(state,path,'trajectory of an explicitly nonaccepted known PSS/Root result; native/recorded cores and final facts scouted separately');
        else
            state=omit(state,path,'literal owning orbit/map core retained; dense trajectory samples replaced by recorded endpoint/event summary');
        end
        % Metadata and termination fields can contain a separately saved
        % primary orbit. Dense numeric time/state tapes remain O(1) leaves.
        for name=properties(v).'
            state=scout(v.(name{1}),[path,'.',name{1}],context,state,depth+1);
        end
    elseif knownRuntime(cls)||isa(v,'function_handle')
        % Preserve recorded policy DATA before omitting the live Map graph.
        if ~isa(v,'function_handle')&&isscalar(v)&&isprop(v,'Map')
            try
                map=v.Map;policy=[];
                if (isobject(map)&&isprop(map,'ReturnPolicy'))||(isstruct(map)&&isfield(map,'ReturnPolicy')),policy=map.ReturnPolicy;end
                if ~isempty(policy)
                    [~,evidence]=RecordedReturnPolicy_v3(struct('correction_return_policy',policy));
                    if evidence.complete,savedPolicy=evidence.configuration;
                    else,[savedPolicy,state]=project(policy,[path,'.recorded_map_policy'],state,depth+1);end
                    state.selected{end+1,1}=struct('path',[path,'.recorded_map_policy'], ...
                        'value',struct('literal_policy_data',savedPolicy,'recorded_policy_evidence',evidence));
                end
            catch problem
                state=unknown(state,path,['recorded runtime-map policy unavailable: ',problem.identifier]);
            end
        end
    elseif isstruct(v)
        names=fieldnames(v);
        for k=1:numel(v)
            here=context;for j=1:numel(names)
                if ismember(names{j},{'report','raw_report','native_report','compacted_report'})
                    c=v(k).(names{j});if isstruct(c)&&isscalar(c),here=c;break;end
                end
            end
            if isfield(v,'schema_version')&&isscalar(v),here=v;end
            if ~knownReport(here)&&knownRoot(v(k)),here=v(k);end
            [mapRecord,mapBound]=recordedMapCore(v(k),here,path);
            if mapBound
                state.owned_trajectories{end+1,1}=sprintf('%s(%d).trajectory',path,k);
                state=scout(mapRecord,[path,'.literal_recorded_map_core'],here,state,depth+1);
            end
            if plainOrbit(v(k))&&~endsWith(path,'.public_properties')
                state.owned_trajectories{end+1,1}=sprintf('%s(%d).trajectory',path,k);
                core=struct();for name={'initial_state','initial_mode','parameter','period','stride_displacement','return_policy','return_policy_name','return_multiplicity','section_chart','event_history','source_parameter_path','source_scope'}
                    if isfield(v,name{1}),core.(name{1})=v(k).(name{1});end
                end
                [~,policy]=RecordedReturnPolicy_v3(here);core.recorded_policy_evidence=policy;
                [core,state]=project(core,[path,'.plain_primary'],state,depth+1);
                record=struct('source_paths',{{path}},'source_array_index',k,'source_container_size',size(v),'source_class','plain recorded orbit core', ...
                    'core29',[v(k).initial_state(:);v(k).stride_displacement(:);v(k).period], ...
                    'core',core,'saved_report_status',reportHead(here));
                key=C1ScientificDigest_v3(rmfield(record,'source_paths'));
                hit=find(strcmp(state.primary_digests,key),1);
                if isempty(hit)
                    assert(numel(state.primary)<state.rules.maximum_primary_records,'AuditFacts:PrimaryCap','Too many primary records; keep raw.');
                    state.primary{end+1,1}=record;state.primary_digests{end+1,1}=key;
                end
            end
            for j=1:numel(names)
                n=names{j};q=v(k).(n);p=sprintf('%s(%d).%s',path,k,n);
                if isa(q,'uint8')&&~isempty(q)&&~isempty(regexpi(n,'(exact_.*file_bytes|checkpoint.*bytes|native.*file_bytes)','once'))
                    state=unknown(state,p,'embedded MAT/ledger image may contain primary state; no implicit binary decode');
                end
                if salient(n)&&plainSmall(q)
                    state.selected{end+1,1}=struct('path',p,'value',q);
                    assert(numel(state.selected)<=20000,'AuditFacts:FactsCap','Too many final diagnostic facts; keep raw.');
                end
                state=scout(q,p,here,state,depth+1);
            end
        end
    elseif iscell(v)
        for k=1:numel(v),state=scout(v{k},sprintf('%s{%d}',path,k),context,state,depth+1);end
    elseif ~(isnumeric(v)||islogical(v)||ischar(v)||isstring(v))
        state=unknown(state,path,['unknown class ',cls]);
    end
end

function [out,state]=project(v,path,state,depth)
    state=visit(state,depth,path);cls=class(v);
    if isa(v,'HybridOrbit_v3')
        out=struct('retained_as','primary_records','source_class',cls,'count',numel(v));
    elseif isa(v,'Trajectory_v3')
        holder=CompactResearchBranch_v3(struct('trajectory',v));out=holder.trajectory;
        [out,state]=project(out,[path,'.trace_summary'],state,depth+1);
    elseif isa(v,'function_handle')
        out=struct('source_class','function_handle','signature',func2str(v),'closure_retained',false,'callable_invoked',false);
        state=omit(state,path,'runtime callback workspace and executable closure');
    elseif knownRuntime(cls)
        data=struct();props=properties(v);
        for k=1:numel(props)
            name=props{k};
            if ismember(name,{'Map','BaseProblem','System','Simulator','Residual','Jacobian','Dynamics','Model'})
                state=omit(state,[path,'.',name],'known runtime method graph');continue
            end
            [data.(name),state]=project(v.(name),[path,'.',name],state,depth+1);
        end
        out=struct('source_class',cls,'recorded_data',data,'runtime_graph_retained',false,'callable_reconstructed',false);
    elseif isstruct(v)
        out=v;names=fieldnames(v);
        for k=1:numel(v),for j=1:numel(names)
            n=names{j};p=sprintf('%s(%d).%s',path,k,n);q=v(k).(n);
            if tape(n)
                out(k).(n)=struct('retained',false,'source_class',class(q),'source_size',size(q),'reason','known dense execution history; primary scout and final-fact selections retained separately');
                state=omit(state,p,'dense trajectory/Root/FD/evaluation/previous-run execution tape');
            else,[out(k).(n),state]=project(q,p,state,depth+1);end
        end,end
    elseif iscell(v)
        out=v;for k=1:numel(v),[out{k},state]=project(v{k},sprintf('%s{%d}',path,k),state,depth+1);end
    elseif isnumeric(v)||islogical(v)
        if issparse(v),state=unknown(state,path,'unsupported sparse storage; retain raw matrix');out=struct('source_class',cls,'source_size',size(v));
        elseif knownModeMask(path,v)&&numel(v)>state.rules.small_numeric_elements
            nz=find(v);out=struct('source_class',cls,'source_size',size(v),'sample_mismatch_count',numel(nz), ...
                'any_sample_mismatch',~isempty(nz),'first_mismatch_linear_indices',nz(1:min(16,end)), ...
                'last_mismatch_linear_indices',nz(max(1,end-15):end),'pointwise_sample_mask_retained',false);
            state=omit(state,path,'known sampled contact-history mismatch mask; exact count/shape/end locations retained');
        elseif numel(v)<=state.rules.small_numeric_elements,out=v;
        elseif primaryTable(path,v)&&numel(v)<=state.rules.primary_table_elements,out=v;
        elseif diagnosticPath(path)
            out=struct('retained',false,'source_class',cls,'source_size',size(v));state=omit(state,path,'dense sampled/trajectory/probe array');
        else,out=struct('retained',false,'source_class',cls,'source_size',size(v));state=unknown(state,path,'ambiguous large numeric array');end
    elseif ischar(v)||isstring(v)
        if numel(v)<=65536,out=v;
        else,out=struct('retained',false,'source_class',cls,'source_size',size(v));state=omit(state,path,'long console/source/debug text');end
    else
        out=struct('retained',false,'source_class',cls,'source_size',size(v));state=unknown(state,path,['unknown object ',cls]);
    end
end

function out=reportHead(r)
    out=struct();if ~isstruct(r)||~isscalar(r),return;end
    for key={'schema','schema_version','status','accepted','acceptance_gates','failure_stage','wrapper_accepted','committed','stage','return_policy_name','correction_return_policy'}
        if isfield(r,key{1})&&plainSmall(r.(key{1})),out.(key{1})=r.(key{1});end
    end
end
function yes=knownRuntime(c)
    yes=ismember(c,{'ConstrainedEnergyResidual_v3','EnergyFamilyResidual_v3','PeriodicOrbitResidual_v3', ...
        'FiniteDifferenceJacobian_v3','ValidSidedFiniteDifferenceJacobian_v3','PoincareMap_v3', ...
        'HybridSimulator_v3','EventDetector_v3','ContinuousDynamics_v3','GuardFunctions_v3','ResetMap_v3', ...
        'Quadrupedal_Dynamics_v3','PoincareSection_v3','BLMarkedApexReturnPolicy_v3', ...

function [out,yes]=recordedMapCore(v,r,path)
    out=struct();yes=false;
    needed={'initial_state','initial_mode','period','stride_displacement'};
    if ~isscalar(v)||~all(isfield(v,needed))||isfield(v,'parameter') ...
            ||isempty(regexpi(path,'(independent_replay|evaluationInfo|lastFiniteEvaluatedBase|lastAcceptedIterate).*map_info','once')) ...
            ||~isempty(regexpi(path,'(iterationJournal|invalidEvaluationDiagnostics|jacobianDiagnostics|phase_results|candidate_section_crossings)','once')),return;end
    parameter=[];origin='';
    if knownReport(r)&&strcmp(r.schema_version,'periodic-solution-solver-v3-2')&&isfield(r,'augmented_parameter')
        parameter=r.augmented_parameter;origin='known PSS report.augmented_parameter first10';
    elseif knownReport(r)&&strcmp(r.schema_version,'root-free-fixed-energy-admission-v1')&&isfield(r,'seed')&&isfield(r.seed,'parameter')
        parameter=r.seed.parameter;origin='known root-free report.seed.parameter';
    elseif knownRoot(r)
        parameter=r.parameter;origin='recorded v3 Root result.parameter first10; no independent model/admission inference';
    end
    if ~isnumeric(parameter)||~any(numel(parameter)==[10,11])||numel(v.initial_state)~=14 ...
            ||numel(v.initial_mode)~=4||~isscalar(v.period)||numel(v.stride_displacement)~=14,return;end
    out=struct('initial_state',v.initial_state,'initial_mode',v.initial_mode, ...
        'parameter',parameter(1:10),'period',v.period,'stride_displacement',v.stride_displacement, ...
        'source_parameter_path',origin,'source_scope','Literal recorded map-return core only; not a new orbit, identity, periodicity, admission or branch claim.');
    for key={'return_policy','return_policy_name','return_multiplicity','section_chart','event_history'}
        if isfield(v,key{1}),out.(key{1})=v.(key{1});end
    end
    yes=true;
end

function o=defaults(v)
    o=struct('MaxWallSeconds',1800,'SaveMarginSeconds',60,'MaximumBytes',32*1024^2,'ResumeReference',struct(),'MaxNewFiles',1024, ...
        'AdoptReviewedCodec',false,'RetryHeldIndices',[]);
    assert(isstruct(v)&&isscalar(v),'AuditFacts:Options','Use a scalar bounded options struct.');
    for key=fieldnames(v).',assert(isfield(o,key{1}),'AuditFacts:Options','Unknown option.');o.(key{1})=v.(key{1});end
    validateattributes(o.MaxWallSeconds,{'double'},{'scalar','real','finite','>',120,'<=',3600});
    validateattributes(o.SaveMarginSeconds,{'double'},{'scalar','real','finite','>=',60,'<',o.MaxWallSeconds});
    validateattributes(o.MaximumBytes,{'double'},{'scalar','real','finite','positive','<=',64*1024^2});
    validateattributes(o.MaxNewFiles,{'double'},{'scalar','real','finite','integer','positive','<=',1024});
    assert(islogical(o.AdoptReviewedCodec)&&isscalar(o.AdoptReviewedCodec),'AuditFacts:Options','Use an explicit logical codec-adoption flag.');
    assert(isnumeric(o.RetryHeldIndices)&&(isvector(o.RetryHeldIndices)||isempty(o.RetryHeldIndices)) ...
        &&all(isfinite(o.RetryHeldIndices(:)))&&all(o.RetryHeldIndices(:)==fix(o.RetryHeldIndices(:))) ...
        &&all(o.RetryHeldIndices(:)>=1&o.RetryHeldIndices(:)<=1024)&&numel(unique(o.RetryHeldIndices))==numel(o.RetryHeldIndices), ...
        'AuditFacts:Options','Retry only explicit unique held-source indices.');
end
function r=seal(r)
    r.file=canonical(r.file);d=dir(r.file);assert(isscalar(d)&&~d.isdir&&d.bytes==r.bytes,'AuditFacts:Source','Roster source is missing or its byte count changed.');
    actual=RoundSHA256_v3(r.file);if isfield(r,'sha256')&&~isempty(r.sha256),assert(strcmp(actual,r.sha256),'AuditFacts:Source','Source digest changed.');end
    r=struct('file',r.file,'sha256',actual,'bytes',d.bytes,'header_scope','Observed immediately before native load; blank inventory SHA is not an earlier digest.');
end
