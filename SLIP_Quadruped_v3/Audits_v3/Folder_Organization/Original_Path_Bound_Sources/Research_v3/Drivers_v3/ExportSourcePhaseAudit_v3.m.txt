function audit=ExportSourcePhaseAudit_v3(state,cfg)
%EXPORTSOURCEPHASEAUDIT_V3 Compact executed source and signed-phase evidence.
% A partial candidate is never counted as a periodic gait or attachment.
    v3=fileparts(fileparts(fileparts(mfilename('fullpath'))));
    output=fullfile(v3,'Research_v3','next_round');rows=struct([]);
    for k=1:numel(state.tasks)
        task=state.tasks(k);
        if ~any(strcmp(task.kind,{'P1_source','P2_source'}))||task.attempts==0,continue;end
        row=struct('task_id',task.id,'round',task.repair_round,'state',task.state, ...
            'scientific_status',task.scientific_status,'attempts',task.attempts, ...
            'charged_wall_seconds',task.wall_seconds,'artifact',task.artifact, ...
            'source_path','','source_variable','','source_columns',0,'selected_columns',[], ...
            'original_density_selected_columns',[],'executed_unique_columns',[], ...
            'untested_columns',[],'accepted_source_records',0,'accepted_gait_labels',{{}}, ...
            'family_sector',struct(),'landing_alternative',struct(),'driver_failure',task.last_failure);
        if ~isempty(task.artifact)&&isfile(fullfile(v3,task.artifact))
            saved=load(fullfile(v3,task.artifact),'result');result=saved.result;
            if isfield(result,'observations')
                row.source_columns=result.source_column_count;row.selected_columns=result.selected_columns;
                row.source_path=result.source.source_path;row.source_variable=result.variable;
                row.original_density_selected_columns=member(result,'original_density_selected_columns',[]);
                row.executed_unique_columns=unique(result.completed_source_columns,'stable');
                row.untested_columns=setdiff(1:row.source_columns,row.executed_unique_columns);
                row.accepted_source_records=result.accepted_count;
                if ~isempty(result.observations)
                    accepted=result.observations([result.observations.accepted]);
                    row.accepted_gait_labels=unique({accepted.gait},'stable');
                end
            end
            if isfield(result,'family_switch')
                phase=result.family_switch;
                row.family_sector=struct('state',phase.state,'phase',phase.phase, ...
                    'parent_independently_admitted',phase.parent_independently_admitted, ...
                    'parent_descriptive_label',phase.parent_descriptive_label, ...
                    'prediction_count',numel(phase.predictions),'next_trial',phase.next_trial, ...
                    'completed_trials',numel(phase.trials),'accepted_orbits',phase.accepted_count, ...
                    'partial_trial',struct(),'ancestry_claimed',false);
                if isfield(phase,'partial_trial')
                    partial=phase.partial_trial;file=fullfile(fileparts(fullfile(v3,task.artifact)),partial.artifact);
                    record=load(file,'report');report=record.report;
                    row.family_sector.partial_trial=struct('prediction_index',partial.prediction_index, ...
                        'signed_amplitude',phase.predictions(partial.prediction_index).signed_amplitude, ...
                        'target',phase.predictions(partial.prediction_index).target_label, ...
                        'artifact',strrep(file,[v3,filesep],''),'attempt',partial.attempt, ...
                        'best_candidate',partial.best_candidate, ...
                        'energy',QuadrupedEnergy_v3.evaluate(partial.best_candidate,phase.parent_seed.mode,phase.parent_seed.parameter), ...
                        'finite_saved_Jacobian',~isempty(partial.final_jacobian)&&all(isfinite(partial.final_jacobian(:))), ...
                        'Jacobian_size',size(partial.final_jacobian), ...
                        'initial_physical_residual',member(report,'initial_physical_norm',NaN), ...
                        'final_physical_residual',member(report,'final_physical_norm',NaN), ...
                        'full_closure',member(report,'full_closure',NaN), ...
                        'primary_failure',report.primary_failure,'replay_failure',report.replay_failure, ...
                        'accepted',false);
                end
            end
            if isfield(result,'landing_alternative')
                p=result.landing_alternative;
                row.landing_alternative=struct('state',p.state,'candidate_count',numel(p.candidates), ...
                    'executed_trials',numel(p.trials),'accepted_count',p.accepted_count,'trials',p.trials);
            end
        end
        if isempty(rows),rows=row;else,rows(end+1)=row;end %#ok<AGROW>
    end
    rounds=struct('id',{'A','B','C'},'source_tasks_executed',{0,0,0});
    if ~isempty(rows)
        for k=1:3,rounds(k).source_tasks_executed=sum(strcmp({rows.round},rounds(k).id));end
    end
    audit=struct('schema_version','executed-source-phase-audit-v3-1', ...
        'generated_utc',char(datetime('now','TimeZone','UTC')),'profile',cfg.profile, ...
        'effective_domain',cfg.domain,'charged_numerical_wall_seconds',state.numerical_wall_seconds, ...
        'registered_total_wall_seconds',cfg.budgets.total_wall_seconds,'rounds',rounds, ...
        'source_tasks',rows,'record_counts_are_distinct_gaits',false, ...
        'partial_candidates_are_accepted',false,'global_completeness_claimed',false);
    RoundSave_v3(fullfile(output,['source_phase_audit_',cfg.profile,'.mat']),struct('audit',audit));
    RoundJSON_v3(fullfile(output,['source_phase_audit_',cfg.profile,'.json']),audit);
end
function value=member(source,name,fallback)
    value=fallback;if isfield(source,name),value=source.(name);end
end
