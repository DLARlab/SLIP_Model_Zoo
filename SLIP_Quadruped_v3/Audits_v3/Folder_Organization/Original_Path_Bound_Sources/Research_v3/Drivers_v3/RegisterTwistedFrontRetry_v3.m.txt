function registration=RegisterTwistedFrontRetry_v3(outputDirectory,previousRegistration)
%REGISTERTWISTEDFRONTRETRY_V3 Standalone preparation; no master/queue writes.
% Calling this function reads the failed pilot and current gates and writes
% only its explicit output directory. It never performs a map evaluation.
    v3=fileparts(fileparts(fileparts(mfilename('fullpath'))));
    if nargin<1,outputDirectory=fullfile(v3,'Research_v3/next_round/solver/twisted_front_retry');end
    if nargin<2,previousRegistration='';end
    previousHash='';driverRevision=1;
    if ~isempty(previousRegistration)
        previousRegistration=V3OutputPath_v3(previousRegistration);
        previous=jsondecode(fileread(previousRegistration));
        assert(strcmp(previous.method,'front-twisted-BL1-common-C1-Broyden-retry'));
        previousHash=RoundSHA256_v3(previousRegistration);driverRevision=2;
    end
    outputDirectory=V3OutputPath_v3(outputDirectory);if ~isfolder(outputDirectory),mkdir(outputDirectory);end
    assert(~isfile(fullfile(outputDirectory,'method_registration.json')) ...
        &&~isfile(fullfile(outputDirectory,'front_retry_candidate.mat')), ...
        'RegisterTwistedFrontRetry_v3:FrozenRegistration', ...
        'Use a new explicit directory to preserve an existing frozen registration.');
    gatePath='Research_v3/next_round/solver/solver_validation_summary.json';
    gate=jsondecode(fileread(fullfile(v3,gatePath)));
    assert(gate.all_required_gates_passed&&numel(gate.cases)>=7&&all([gate.cases.passed]), ...
        'RegisterTwistedFrontRetry_v3:SolverGate','Current canonical seven-case solver validation must pass.');
    checkHashes(gate.source_hashes,'canonical solver gate');
    metadataPath='Research_v3/next_round/solver/twisted_return/Broyden_metadata_gate.json';
    metadata=jsondecode(fileread(fullfile(v3,metadataPath)));
    assert(metadata.passed&&metadata.missing_event_history_rejected&&metadata.raw_one_return_preserved ...
        &&metadata.map_info_unchanged&&metadata.rejected_raw_signatures_preserved, ...
        'RegisterTwistedFrontRetry_v3:MetadataGate','Verified actual metadata and missing-event negative control must pass.');
    assert(strcmp(metadata.adapter_sha256,RoundSHA256_v3(fullfile(v3,'Orbit_v3/TwistedReturnFamilyResidual_v3.m'))) ...
        &&strcmp(metadata.helper_sha256,RoundSHA256_v3(fullfile(v3,'Numerics_v3/EventOrderChartJacobian_v3.m'))));
    pilotPath='Research_v3/next_round/solver/twisted_return/pilot.mat';
    frozenPath='Research_v3/next_round/solver/twisted_return/registration.mat';
    loaded=load(fullfile(v3,pilotPath),'report','summary');old=loaded.report;
    frozen=load(fullfile(v3,frozenPath),'constraint','registration');
    assert(~old.accepted&&numel(old.final_coordinates)==13&&any(~isfinite(old.solver.finalJacobian(:))), ...
        'RegisterTwistedFrontRetry_v3:OriginalPilot','Expected the diagnosed unaccepted pilot and partial final Jacobian.');
    metadataRegistration=jsondecode(fileread(fullfile(v3,'Research_v3/next_round/solver/twisted_return/Broyden_metadata_registration.json')));
    assert(strcmp(metadataRegistration.source_pilot_sha256,RoundSHA256_v3(fullfile(v3,pilotPath))), ...
        'RegisterTwistedFrontRetry_v3:PilotHash','The metadata gate must bind this exact failed pilot.');
    oldHashes=loaded.summary.source_hashes;
    enginePaths={'Orbit_v3/PoincareMap_v3.m','Simulation_v3/HybridSimulator_v3.m','Simulation_v3/EventDetector_v3.m'};
    physicalEngine=oldHashes(ismember({oldHashes.path},enginePaths));
    assert(numel(physicalEngine)==3);checkHashes(physicalEngine,'original pilot physical engine');
    modelPath='Research_v3/next_round/opposed_spread_period_two_registration.json';
    model=jsondecode(fileread(fullfile(v3,modelPath)));checkHashes(model.engine_hashes,'completed production-model audit');
    original=old.options.Residual;w=old.final_coordinates(:);q=old.seed.mode(:);p=old.seed.parameter(:);
    assert(isequal(frozen.constraint,original.Constraint)&&isequal(p,original.OneFamily.PhysicalParameter));
    integration=frozen.registration.integration;
    simulator=HybridSimulator_v3(integration);
    assert(isequal(simulator.Options,original.Map.Simulator.Options), ...
        'RegisterTwistedFrontRetry_v3:Integration', ...
        'The frozen original integration must reconstruct the same simulator settings.');
    reducedTolerance=min(1e-10,min(old.acceptance.FullClosureTolerance,old.acceptance.EnergyTolerance)/100);
    functionTolerance=1e-11;
    if isfield(old.options.SolverOptions,'ResidualAcceptanceTolerance')
        reducedTolerance=min(reducedTolerance,old.options.SolverOptions.ResidualAcceptanceTolerance);
    end
    if isfield(old.options.SolverOptions,'FunctionTolerance')
        functionTolerance=min(functionTolerance,old.options.SolverOptions.FunctionTolerance);
    end
    functionTolerance=min(functionTolerance,reducedTolerance/10);
    candidate=struct('id','n0_positive_front_twisted_common_C1_Broyden_retry', ...
        'seed',old.seed,'coordinates',w,'reference_state',original.OneFamily.Chart.reference_state, ...
        'coordinate_indices',original.OneFamily.CoordinateIndices,'constraint',frozen.constraint, ...
        'physical_parameter',p,'mode',q,'original_pilot',pilotPath,'original_registration',frozenPath, ...
        'original_source_hashes',oldHashes,'original_initial_residual',old.initial_scaled_norm, ...
        'original_final_residual',old.final_scaled_norm,'initial_jacobian',[], ...
        'ancestry_claimed',false,'unrestricted_C2_claimed',false);
    candidate.seed.state=old.final_candidate;candidate.seed.occurrence=2;
    candidate.seed.return_policy='BL-marked-apex-return';
    candidate.seed.provenance=struct('kind','prospective-diagnosed-front-C1-Broyden-retry', ...
        'original_pilot',pilotPath,'original_pilot_sha256',RoundSHA256_v3(fullfile(v3,pilotPath)), ...
        'fixed_signed_amplitude',frozen.constraint.target,'no_saved_partial_J_reuse',true);
    candidateFile=fullfile(outputDirectory,'front_retry_candidate.mat');
    RoundSave_v3(candidateFile,struct('candidate',candidate));
    paths={'Orbit_v3/TwistedReturnFamilyResidual_v3.m','Orbit_v3/EnergyFamilyResidual_v3.m', ...
        'Numerics_v3/PeriodicSolutionSolver_v3.m','Numerics_v3/RootSolver_v3.m', ...
        'Numerics_v3/EventOrderChartJacobian_v3.m','Research_v3/Drivers_v3/RunTwistedFrontRetry_v3.m', ...
        'Research_v3/Drivers_v3/RegisterTwistedFrontRetry_v3.m'};
    hashes=struct('path',{},'sha256',{});
    for k=1:numel(paths)
        hashes(k)=struct('path',paths{k},'sha256',RoundSHA256_v3(fullfile(v3,paths{k})));
    end
    methodFile=fullfile(outputDirectory,'method_registration.json');
    record=struct('id',candidate.id,'candidate_artifact',relative(candidateFile),'candidate_variable','candidate', ...
        'candidate_sha256',RoundSHA256_v3(candidateFile),'chart_audit',metadataPath, ...
        'method','front-twisted-BL1-common-C1-Broyden-retry','method_registration',relative(methodFile));
    registration=struct('schema_version','prospective-native-front-twisted-retry-v3-2', ...
        'registered_utc',char(datetime('now','TimeZone','UTC')),'matlab_version',version, ...
        'method',record.method,'candidates',record,'source_hashes',hashes, ...
        'driver_revision',driverRevision,'supersedes_registration',previousRegistration, ...
        'supersedes_registration_sha256',previousHash, ...
        'revision_scope','Compatible best-iterate partial resume, cumulative cost accounting and separate exact-p10 campaign-domain admission; unchanged objective, physical model and acceptance thresholds.', ...
        'canonical_gate',gatePath,'canonical_gate_sha256',RoundSHA256_v3(fullfile(v3,gatePath)), ...
        'canonical_source_hashes',gate.source_hashes,'metadata_gate',metadataPath, ...
        'metadata_gate_sha256',RoundSHA256_v3(fullfile(v3,metadataPath)), ...
        'original_pilot',pilotPath,'original_pilot_sha256',RoundSHA256_v3(fullfile(v3,pilotPath)), ...
        'original_source_hashes',oldHashes,'original_physical_engine_hashes',physicalEngine, ...
        'production_model_audit_registration',modelPath,'physical_model_hashes',model.engine_hashes, ...
        'epoch_qualification','Original pilot recorded the three engine hashes above; additional production model hashes come from the completed later actual-model audit, not retroactive original-pilot hashes.', ...
        'integration',integration, ...
        'replay_integration',frozen.registration.replay_integration,'acceptance',old.acceptance, ...
        'constraint',frozen.constraint,'fd_relative_candidate_steps',[3e-4,1e-4], ...
        'fd_maximum_relative_error',.01,'fd_use_richardson',false, ...
        'maximum_correction_wall_seconds',480,'maximum_newton_iterations',25,'maximum_function_evaluations',1000, ...
        'initial_J_reuse',false,'CommonC1BroydenMetadata',true,'Jacobian_refresh_accepted_steps',5, ...
        'function_tolerance',functionTolerance,'residual_acceptance_tolerance',reducedTolerance, ...
        'diagnosis','Original second FD exhausted its slice after raw TD cluster merging invalidated Broyden; actual regular C1 chart permits verified compatibility normalization.', ...
        'Newton_policy','Actual BL1 twisted return with fresh full13D derivative, frozen signed amplitude and free internal E.', ...
        'acceptance_policy','Actual physical BL2 orbit and new tighter shared-service replay; unchanged full-state/energy/stance/primitive gates.', ...
        'standalone_preparation_only',true,'active_queue_modified',false,'numerics_executed',false, ...
        'acceptance_tolerances_changed',false,'target_gait_claimed',false,'ancestry_claimed',false,'unrestricted_C2_claimed',false);
    RoundSave_v3(fullfile(outputDirectory,'method_registration.mat'),struct('registration',registration));
    RoundJSON_v3(methodFile,registration);
    function checkHashes(items,scope)
        for j=1:numel(items)
            assert(strcmp(items(j).sha256,RoundSHA256_v3(fullfile(v3,items(j).path))), ...
                'RegisterTwistedFrontRetry_v3:SourceEpoch','Source changed since %s: %s.',scope,items(j).path);
        end
    end
    function path=relative(file),path=strrep(file,[v3,filesep],'');end
end
