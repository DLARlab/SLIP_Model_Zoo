function audit=ValidateGUIBL2Occurrence_v3(outputDirectory)
%VALIDATEGUIBL2OCCURRENCE_V3 Actual GUI correction preserves a repeated return.
    v3=fileparts(fileparts(fileparts(fileparts(mfilename('fullpath')))));addpath(genpath(v3));
    if nargin<1,outputDirectory=fileparts(mfilename('fullpath'));end
    outputDirectory=V3OutputPath_v3(outputDirectory);timer=tic;
    inputPath=fullfile(v3,'Research_v3','next_round','solver','case6_marked_occurrence.mat');
    input=load(inputPath,'seed','report','result');
    orbit=input.report.solver.orbit;
    assert(isa(orbit,'HybridOrbit_v3')&&orbit.section_chart.BL_touchdowns_per_return==2, ...
        'Use the actual saved, numerically closed artificial two-cycle PIP orbit.');
    seedFile=fullfile(outputDirectory,'GUI_BL2_occurrence_seed.mat');
    save(seedFile,'orbit','-v7');
    controller=QuadrupedGUIController_v3(v3);controller.CorrectionSymmetry='pronk';
    dataset=controller.loadDataset(seedFile);controller.select(dataset,1);
    selectedBefore=controller.selected();
    assert(strcmp(controller.ReturnPolicyOverride,'stored') ...
        &&selectedBefore.BL_touchdowns_per_return==2&&~selectedBefore.accepted);
    record=controller.correct(selectedBefore);
    selectedAfter=controller.selected();candidate=controller.Candidate;
    report=record.validation;
    declared=report.seed.occurrence;
    correctionOccurrence=report.solver.orbit.section_chart.BL_touchdowns_per_return;
    replayOccurrence=report.independent_replay.map_info.section_chart.BL_touchdowns_per_return;
    primitiveStatus=string(report.primitive.status);
    audit=struct('schema_version','GUI-BL2-occurrence-audit-v3-1','passed',false, ...
        'matlab_version',version,'platform',computer, ...
        'input_artifact',inputPath,'input_sha256',RoundSHA256_v3(inputPath), ...
        'stored_selected_occurrence',selectedBefore.BL_touchdowns_per_return, ...
        'GUI_override',controller.ReturnPolicyOverride,'GUI_default_occurrence_option',controller.ReturnOccurrences, ...
        'service_declared_occurrence',declared,'correction_occurrence',correctionOccurrence, ...
        'independent_replay_occurrence',replayOccurrence,'primitive_status',primitiveStatus, ...
        'full_closure',report.full_closure,'service_accepted',report.accepted, ...
        'returned_record_accepted',record.accepted,'selected_record_accepted',selectedAfter.accepted, ...
        'failed_candidate_retained',isstruct(candidate)&&isfield(candidate,'validation') ...
        &&numel(candidate.initial_state)==14&&~candidate.accepted, ...
        'failed_report_retained',isfield(candidate.validation,'primitive') ...
        &&isfield(candidate.validation,'independent_replay')&&isfield(candidate.validation,'solver'), ...
        'GUI_status',controller.Status,'elapsed_seconds',toc(timer), ...
        'GUIController_sha256',RoundSHA256_v3(fullfile(v3,'GUI_v3','QuadrupedGUIController_v3.m')), ...
        'PeriodicSolutionSolver_sha256',RoundSHA256_v3(fullfile(v3,'Numerics_v3','PeriodicSolutionSolver_v3.m')), ...
        'PoincareMap_sha256',RoundSHA256_v3(fullfile(v3,'Orbit_v3','PoincareMap_v3.m')), ...
        'HybridSimulator_sha256',RoundSHA256_v3(fullfile(v3,'Simulation_v3','HybridSimulator_v3.m')));
    audit.passed=declared==2&&correctionOccurrence==2&&replayOccurrence==2 ...
        &&primitiveStatus=="multiple_cover_detected"&&report.full_closure<=1e-8 ...
        &&~report.accepted&&~record.accepted&&~selectedAfter.accepted ...
        &&audit.failed_candidate_retained&&audit.failed_report_retained;
    save(fullfile(outputDirectory,'GUI_BL2_occurrence_audit.mat'), ...
        'audit','selectedBefore','selectedAfter','record','candidate','-v7');
    RoundJSON_v3(fullfile(outputDirectory,'GUI_BL2_occurrence_audit.json'),audit);disp(audit);
    assert(audit.passed,'Actual GUI correction must preserve BL2 and reject the repeated PIP cycle.');
end
