function summary=ValidateSphereSecondSeed_v3(outputDirectory)
%VALIDATESPHERESECONDSEED_V3 Preserve GUI radius and shared fresh acceptance.
    v3=fileparts(fileparts(fileparts(fileparts(mfilename('fullpath')))));
    addpath(genpath(v3));if nargin<1,outputDirectory=fileparts(mfilename('fullpath'));end
    outputDirectory=V3OutputPath_v3(outputDirectory);
    directFile=fullfile(outputDirectory,'sphere_second_seed.mat');
    if ~isfile(directFile)
        [first,~,framework]=QuadrupedalExample_v3(struct('Refine',false));radius=2e-4;
        options=struct('Symmetry','pronk','SolverOptions',solverOptions());
        [orbit,report]=FixedParameterContinuation_v3.secondSeed(framework.residual, ...
            framework.residual.packState(first.initial_state),first.parameter,first.initial_mode,radius,options);
        save(directFile,'first','radius','options','orbit','report','-v7');
    end
    direct=load(directFile);
    controller=QuadrupedGUIController_v3(v3);controller.SolverOptions=solverOptions();
    controller.loadDataset(fullfile(v3,'P1_Breaking_Symmetries_Leads_to_Diverse_Qudrupedal_Gaits', ...
        'SourceFixtures_v3','PK_20_2.mat'));
    record=controller.secondSeed(direct.radius,'pronk');
    summary=struct('schema_version','focused-sphere-second-seed-validation-v3-1', ...
        'passed',false,'requested_scaled_radius',direct.radius, ...
        'actual_scaled_radius',direct.report.actual_scaled_distance, ...
        'radius_equation_error',direct.report.actual_scaled_distance-direct.radius, ...
        'fresh_full_closure',direct.report.periodic_solution_report.full_closure, ...
        'accepted_Newton_iterations',direct.report.acceptedNewtonIterations, ...
        'GUI_accepted',record.accepted,'GUI_scaled_radius',record.provenance.scaled_state_distance, ...
        'GUI_radius_error',record.provenance.scaled_state_distance-direct.radius, ...
        'GUI_replay_norm',record.residual_norm,'GUI_status',controller.Status, ...
        'sphere_adapter_sha256',RoundSHA256_v3(fullfile(v3,'Orbit_v3','SphereFamilyResidual_v3.m')), ...
        'family_service_sha256',RoundSHA256_v3(fullfile(v3,'Numerics_v3','FixedParameterContinuation_v3.m')));
    summary.passed=direct.report.converged&&record.accepted ...
        &&abs(summary.radius_equation_error)<=1e-8&&abs(summary.GUI_radius_error)<=1e-8 ...
        &&summary.fresh_full_closure<=1e-8&&summary.accepted_Newton_iterations>0;
    save(fullfile(outputDirectory,'sphere_GUI_validation.mat'),'record','summary','-v7');
    f=fopen(fullfile(outputDirectory,'sphere_GUI_validation.json'),'w');c=onCleanup(@()fclose(f));
    fprintf(f,'%s\n',jsonencode(summary,'PrettyPrint',true));disp(summary);
end
function options=solverOptions()
    options=struct('Algorithm','newton','MaxIterations',25,'MaxFunctionEvaluations',600, ...
        'Jacobian',HybridFiniteDifferenceJacobian_v3(struct('RelativeCandidateSteps',[3e-5,1e-5])));
end
