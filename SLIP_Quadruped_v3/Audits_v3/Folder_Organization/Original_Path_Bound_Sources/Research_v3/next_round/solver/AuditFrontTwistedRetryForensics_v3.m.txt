function audit=AuditFrontTwistedRetryForensics_v3(outputDirectory)
%AUDITFRONTTWISTEDRETRYFORENSICS_V3 Saved report algebra; zero map calls.
    v3=fileparts(fileparts(fileparts(fileparts(mfilename('fullpath')))));
    if nargin<1,outputDirectory=fullfile(v3,'Research_v3/next_round/solver/front_retry_forensics');end
    outputDirectory=V3OutputPath_v3(outputDirectory);if ~isfolder(outputDirectory),mkdir(outputDirectory);end
    file='Research_v3/next_round/tasks/full/n0_positive_front_twisted_common_C1_Broyden_retry/front_retry_report.mat';
    hashBefore=RoundSHA256_v3(fullfile(v3,file));loaded=load(fullfile(v3,file),'report','result');report=loaded.report;
    assert(~report.accepted);J=report.solver.finalJacobian;r=report.solver.residual(:);q=report.solver.jacobianDiagnostics;
    assert(isequal(size(J),[13,13])&&all(isfinite(J(:)))&&all(report.solver.perColumnReliability));
    [U,S,V]=svd(J);singular=diag(S);step=-J\r;scale=report.solver.stateScale(:);
    relative=q.estimatedErrors(:);absolute=zeros(13,1);
    for k=1:13
        % UseRichardson=false: selected derivative is the fine estimate.
        fine=q.columns(k).derivativeEstimate(:);
        absolute(k)=relative(k)*max(1,norm(fine))/scale(k);
    end
    assert(strcmp(hashBefore,RoundSHA256_v3(fullfile(v3,file))));
    info=dir(fullfile(v3,file));stamp=datetime(info.datenum,'ConvertFrom','datenum','TimeZone','local');stamp.TimeZone='UTC';
    gatePath='Research_v3/next_round/solver/solver_validation_summary.json';gate=jsondecode(fileread(fullfile(v3,gatePath)));
    for k=1:numel(gate.source_hashes)
        item=gate.source_hashes(k);assert(strcmp(item.sha256,RoundSHA256_v3(fullfile(v3,item.path))));
    end
    methodPath='Research_v3/next_round/solver/twisted_front_retry_02/method_registration.json';method=jsondecode(fileread(fullfile(v3,methodPath)));
    directionSource='Research_v3/next_round/solver/attachment_accuracy_repair/PIP_candidate_1/input_snapshot.mat';
    frozen=load(fullfile(v3,directionSource),'originalRecord');direction=frozen.originalRecord.critical_direction(:);
    physicalDirection=zeros(14,1);physicalDirection(2)=direction(1);
    physicalDirection([7,9,11,13])=direction(2);physicalDirection([8,10,12,14])=direction(3);
    pronkDirection=[physicalDirection([2,3,5,6,7:14]);0];pronkDirection=pronkDirection/norm(pronkDirection);
    scaledGradient=(J.*scale.').'*r;
    audit=struct('schema_version','saved-front-twisted-retry-forensics-v3-1', ...
        'completed_utc',char(datetime('now','TimeZone','UTC')),'matlab_version',version, ...
        'actual_report',file,'actual_report_sha256',hashBefore,'actual_report_modified_utc',char(stamp), ...
        'method_registration',methodPath,'method_registration_sha256',RoundSHA256_v3(fullfile(v3,methodPath)), ...
        'canonical_gate',gatePath,'canonical_gate_sha256',RoundSHA256_v3(fullfile(v3,gatePath)), ...
        'canonical_source_hashes',gate.source_hashes,'corrector_source_hashes',method.source_hashes, ...
        'accepted',report.accepted,'wrapper_state',loaded.result.state,'primary_failure',report.primary_failure, ...
        'finalization_failure',report.finalization_failure,'replay_failure',report.replay_failure, ...
        'initial_twisted_norm',report.initial_scaled_norm,'final_twisted_norm',report.final_scaled_norm, ...
        'actual_BL2_closure',report.full_closure,'full_closure_tolerance',report.acceptance.FullClosureTolerance, ...
        'independent_replay_performed',~isempty(fieldnames(report.independent_replay)), ...
        'function_evaluations',report.solver.functionEvaluationCount,'core_map_evaluations',report.solver.mapEvaluationCount, ...
        'Jacobian_evaluations',report.solver.jacobianEvaluationCount,'accepted_Newton_steps',report.solver.acceptedNewtonIterations, ...
        'invalid_evaluations',report.solver.invalidEvaluationCount,'elapsed_seconds',report.elapsed_seconds, ...
        'all_13_columns_reliable',all(report.solver.perColumnReliability),'common_C1_chart',q.commonC1EventOrderChart, ...
        'singular_values',singular,'condition_estimate',singular(1)/singular(end), ...
        'left_residual_coordinates',U.'*r,'weakest_left_residual_fraction',(U(:,end).'*r)^2/(r.'*r), ...
        'two_weakest_right_vectors',V(:,end-1:end),'reported_firstorderopt',report.solver.output.firstorderopt, ...
        'pronk_direction_source',directionSource,'pronk_direction_source_sha256',RoundSHA256_v3(fullfile(v3,directionSource)), ...
        'measured_pronk_direction_alignment',abs(pronkDirection.'*V(:,end)), ...
        'physical_coordinate_J_transpose_r',J.'*r,'physical_gradient_inf_norm',norm(J.'*r,inf), ...
        'root_coordinate_scale',scale,'scaled_gradient_inf_norm',norm(scaledGradient,inf), ...
        'reported_column_relative_refinement_errors',relative,'estimated_absolute_column_differences',absolute, ...
        'estimated_matrix_difference_Frobenius',norm(absolute),'smallest_singular_value_resolved',false, ...
        'formal_undamped_Newton_step',step,'formal_step_norm',norm(step),'linear_backward_residual',norm(J*step+r,inf), ...
        'accepted_scaled_step_norm',report.solver.output.stepsize, ...
        'Broyden_denominator_squared_step',report.solver.output.stepsize^2,'Broyden_required_squared_step',eps, ...
        'Broyden_small_step_guard_fails',report.solver.output.stepsize^2<=eps, ...
        'rank_scope','Finite reliable columns do not resolve the weakest singular value below empirical FD refinement error; no rank or inverse-error certificate.', ...
        'recommendation','Preserve unresolved near-null front result; do not presume an optimality-only warm retry is cheap or resolves the common-pronk direction.', ...
        'new_front_retry_prepared',false,'return_map_evaluations_in_forensics',0,'core_or_active_driver_changed',false);
    RoundSave_v3(fullfile(outputDirectory,'audit.mat'),struct('audit',audit,'J',J,'residual',r));
    RoundJSON_v3(fullfile(outputDirectory,'audit.json'),audit);
    fprintf('Finite13D cond%.6g sigma_min%.6g scaled_gradient%.6g relative_FD%.6g absolute_matrix_est%.6g formal_step%.6g\n', ...
        audit.condition_estimate,singular(end),audit.scaled_gradient_inf_norm,max(relative),norm(absolute),norm(step));
end
