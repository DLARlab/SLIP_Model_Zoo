function summary=AuditInitialJacobianReuse_v3(outputDirectory)
%AUDITINITIALJACOBIANREUSE_V3 Paired actual-model warm/cold correction audit.
    v3=fileparts(fileparts(fileparts(fileparts(mfilename('fullpath')))));addpath(genpath(v3));
    if nargin<1,outputDirectory=fullfile(fileparts(mfilename('fullpath')),'initial_jacobian_reuse');end
    outputDirectory=V3OutputPath_v3(outputDirectory);if ~isfolder(outputDirectory),mkdir(outputDirectory);end
    sourcePath='Research_v3/next_round/solver/validation_05/case2_actual_correction.mat';
    source=load(fullfile(v3,sourcePath),'corrected','correctionReport');previous=source.correctionReport;
    assert(previous.accepted&&all(isfinite(previous.solver.finalJacobian(:))));
    seed=struct('state',source.corrected.initial_state,'mode',source.corrected.initial_mode, ...
        'parameter',source.corrected.parameter,'return_policy','BL-marked-apex-return','occurrence',1, ...
        'provenance',struct('kind','focused-compatible-initial-J-neighbor','source_artifact',sourcePath));
    seed.state(2)=seed.state(2)+5e-5;
    options=struct('Symmetry','pronk','Energy',previous.augmented_parameter(11),'MaxWallSeconds',180, ...
        'SolverOptions',struct('Algorithm','newton','MaxIterations',20,'MaxFunctionEvaluations',600, ...
        'UseBroyden',true,'JacobianRefreshInterval',5,'Jacobian', ...
        HybridFiniteDifferenceJacobian_v3(struct('RelativeCandidateSteps',[3e-5,1e-5],'MaximumRelativeError',.05))));
    J=previous.solver.finalJacobian;assert(isequal(size(J),[4,4]));
    registration=struct('schema_version','prospective-actual-initial-J-reuse-audit-v3-1', ...
        'frozen_utc',char(datetime('now','TimeZone','UTC')),'matlab_version',version, ...
        'source_artifact',sourcePath,'source_sha256',RoundSHA256_v3(fullfile(v3,sourcePath)), ...
        'seed_state',seed.state,'parameter',seed.parameter,'mode',seed.mode, ...
        'energy',options.Energy,'InitialJacobian',J,'perturbation_in_dx',5e-5, ...
        'compatibility','Same fixed p10, q, energy, pronk chart, four residual equations, integration and BL1 marking as the validated source.', ...
        'root_sha256',RoundSHA256_v3(fullfile(v3,'Numerics_v3','RootSolver_v3.m')), ...
        'required','Both accept fresh physics; warm uses its initial J, performs meaningful Newton steps, and uses fewer new FD evaluations.', ...
        'acceptance_tolerances_changed',false);
    save(fullfile(outputDirectory,'registration.mat'),'registration','seed','options','-v7');
    RoundJSON_v3(fullfile(outputDirectory,'registration.json'),registration);
    warmOptions=options;warmOptions.SolverOptions.ReuseJacobian=true;warmOptions.SolverOptions.InitialJacobian=J;
    [warmOrbit,warmReport]=PeriodicSolutionSolver_v3(seed,warmOptions);
    save(fullfile(outputDirectory,'warm.mat'),'warmOrbit','warmReport','warmOptions','-v7');
    [coldOrbit,coldReport]=PeriodicSolutionSolver_v3(seed,options);
    save(fullfile(outputDirectory,'cold.mat'),'coldOrbit','coldReport','options','-v7');
    summary=struct('schema_version','actual-initial-J-reuse-audit-v3-1','passed',false, ...
        'completed_utc',char(datetime('now','TimeZone','UTC')),'root_sha256',registration.root_sha256, ...
        'warm',compact(warmReport),'cold',compact(coldReport),'state_difference',Inf);
    if warmReport.accepted&&coldReport.accepted
        summary.state_difference=norm(warmOrbit.initial_state-coldOrbit.initial_state,inf);
    end
    summary.passed=warmReport.accepted&&coldReport.accepted&&summary.warm.initial_J_used ...
        &&~summary.cold.initial_J_used&&summary.warm.Newton_steps>0&&warmReport.initial_scaled_norm>1e-6 ...
        &&summary.warm.new_Jacobian_evaluations<summary.cold.new_Jacobian_evaluations ...
        &&summary.state_difference<=1e-8;
    save(fullfile(outputDirectory,'summary.mat'),'summary','-v7');
    RoundJSON_v3(fullfile(outputDirectory,'summary.json'),summary);disp(summary);
    assert(summary.passed,'AuditInitialJacobianReuse_v3:Failed','Paired actual-model warm-J audit failed.');
    function result=compact(report)
        result=struct('accepted',report.accepted,'initial_residual',report.initial_scaled_norm, ...
            'final_residual',report.final_scaled_norm,'fresh_full_closure',report.full_closure, ...
            'elapsed_seconds',report.elapsed_seconds,'primary_failure',report.primary_failure, ...
            'finalization_failure',report.finalization_failure,'replay_failure',report.replay_failure);
        result.initial_J_used=report.solver.output.initialJacobianUsed;
        result.new_Jacobian_evaluations=report.solver.jacobianEvaluationCount;
        result.core_objective_map_count=report.solver.mapEvaluationCount;
        result.Newton_steps=report.solver.acceptedNewtonIterations;
        result.derivative_evaluated=report.solver.derivativeEvaluated;
    end
end
