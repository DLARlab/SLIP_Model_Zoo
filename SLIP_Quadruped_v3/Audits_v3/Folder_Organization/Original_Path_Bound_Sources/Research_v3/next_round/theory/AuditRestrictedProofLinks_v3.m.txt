function audit=AuditRestrictedProofLinks_v3()
%AUDITRESTRICTEDPROOFLINKS_V3 Read-only proof/document-link review; no maps.
    clock=tic;here=fileparts(mfilename('fullpath'));v3=fileparts(fileparts(fileparts(here)));
    addpath(fullfile(v3,'Research_v3/Drivers_v3'));
    documents={'Restricted_Odd_Quarter_PIP_Existence_v3.md', ...
        'Restricted_PIP_Opposed_Spread_Period_Two_v3.md', ...
        'Low_Energy_PIP_Critical_Independent_Audit_v3.md', ...
        'Restricted_PIP_Parent_Variational_Map_v3.md','Next_Round_Mathematical_Audit_v3.md', ...
        'Low_Energy_PK_Bridge_Experiment_v3.md','Restricted_Existence_Final_Scientific_Review_v3.md'};
    records=struct([]);missing={};external={};localCount=0;
    for k=1:numel(documents)
        file=fullfile(v3,'Docs_v3',documents{k});content=fileread(file);
        tokens=regexp(content,'\[[^\]]*\]\(([^\)]+)\)','tokens');
        for j=1:numel(tokens)
            link=tokens{j}{1};if startsWith(link,'http'),external{end+1}=link;continue;end %#ok<AGROW>
            link=regexprep(link,'#.*$','');link=regexprep(link,'^<|>$','');
            if isempty(link),continue;end
            target=fullfile(fileparts(file),link);localCount=localCount+1;
            % This JSON record is written below; its forward link is allowed.
            self=strcmp(link,'../Research_v3/next_round/theory/restricted_proof_links_review.json');
            if ~isfile(target)&&~isfolder(target)&&~self,missing{end+1}=[documents{k},': ',link];end %#ok<AGROW>
        end
        item=struct('document',['Docs_v3/',documents{k}],'sha256',RoundSHA256_v3(file),'markdown_links',numel(tokens));
        if isempty(records),records=item;else,records(end+1)=item;end %#ok<AGROW>
    end
    exact=1+(2*(0:2)+1).^2*pi^2/160;
    assert(abs(exact(3)-1.6959508361671367)>.8);
    audit=struct('schema_version','restricted-proof-scope-and-link-review-v3-1', ...
        'created_utc',char(datetime('now','TimeZone','UTC')),'documents',records, ...
        'local_links_checked',localCount,'missing_local_links',{missing}, ...
        'external_primary_links',{unique(external)},'external_links_not_filesystem_checked',true, ...
        'exact_odd_quarter_energies_n0_n1_n2',exact, ...
        'additional_numerical_candidates',[1.1171137022972106,1.6959508361671367], ...
        'numerical_candidate2_is_exact_n2',false,'unrestricted_C2_supplied_by_restricted_proofs',false, ...
        'named_low_root_interval_certified',false,'opposite_signed_standing_states_are_half_period_phases',true, ...
        'automatic_requested_B2_F2_H2_G2_assignment',false,'final_campaign_snapshot_reviewed',false, ...
        'new_maps',0,'new_simulations',0,'elapsed_seconds',toc(clock),'passed',isempty(missing));
    RoundJSON_v3(fullfile(here,'restricted_proof_links_review.json'),audit);
    assert(audit.passed,'RestrictedProofReview:MissingLinks','Reviewed proof documents contain missing local links.');
    fprintf('Reviewed %d proof documents and %d local links; no missing links, no maps.\n',numel(documents),localCount);
end
