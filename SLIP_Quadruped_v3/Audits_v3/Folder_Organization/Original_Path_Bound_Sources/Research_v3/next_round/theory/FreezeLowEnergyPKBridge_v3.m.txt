function registration=FreezeLowEnergyPKBridge_v3()
%FREEZELOWENERGYPKBRIDGE_V3 Immutable bridge equations and inputs; no maps.
    here=fileparts(mfilename('fullpath'));v3=fileparts(fileparts(fileparts(here)));
    folders={'Schema_v3','Adapters_v3','Dynamics_v3','Simulation_v3','Orbit_v3', ...
        'Numerics_v3','Stability_v3','Research_v3/Drivers_v3'};
    for k=1:numel(folders),addpath(fullfile(v3,folders{k}));end
    registrationPath='Research_v3/next_round/low_energy_PK_bridge_registration.json';
    assert(~isfile(fullfile(v3,registrationPath)),'Frozen bridge registration already exists.');
    snapshot='Research_v3/next_round/theory/low_energy_critical_20261007T143707040/';
    inputs={[snapshot,'01_phase_checkpoint.mat'],[snapshot,'02_family_-1.mat'], ...
        [snapshot,'11_daughter_-0.01_attempt_1.mat'],[snapshot,'audit.mat']};
    expected={'3fd079363512cc79acff3695247ffb36e5c936951ab34308a2eeb9592ecf81fd', ...
        'd4bac9d56c08d2388f96f3d783d42ee39f2da0009710f35356f0b7c030be0e72', ...
        '79fd06d3fa4861e1eec34c605695985993838fe3a0f1d6bdfeeff9b56c9d887d', ...
        'ab3af0f9bcb97db7acefdb368f58ce4f0035a59ba814a3ff66ea2d1511e68819'};
    for k=1:numel(inputs),assert(strcmp(expected{k},RoundSHA256_v3(fullfile(v3,inputs{k}))));end
    loaded=load(fullfile(v3,inputs{1}),'record');parentRecord=loaded.record;
    loaded=load(fullfile(v3,inputs{2}),'branch');sourceOrbit=loaded.branch.points(2).orbit;
    loaded=load(fullfile(v3,inputs{3}),'entry','orbit');targetEntry=loaded.entry;targetOrbit=loaded.orbit;
    loaded=load(fullfile(v3,inputs{4}),'audit');assert(loaded.audit.passed);
    assert(targetEntry.accepted&&targetEntry.signed_amplitude==-.01);
    p=[10;10;20;20;1;1;0;0;2;.5];q=false(4,1);
    assert(isequal(sourceOrbit.parameter(:),p)&&isequal(targetOrbit.parameter(:),p) ...
        &&~any(sourceOrbit.initial_mode)&&~any(targetOrbit.initial_mode));
    sourceState=sourceOrbit.initial_state(:);seedState=sourceState;
    seedState([1,4,5,6])=0;seedState([9,11,13])=seedState(7);seedState([10,12,14])=seedState(8);
    projectionError=norm(seedState-sourceState,inf);
    assert(projectionError<1e-10,'Source point requires more than roundoff canonicalization.');
    sourceClosure=norm(sourceOrbit.poincare_state(2:end)-sourceState(2:end),inf);assert(sourceClosure<1e-8);
    parentState=zeros(14,1);parentState(3)=parentRecord.critical_energy;
    direction=parentRecord.critical_direction(:);assert(numel(direction)==3&&abs(norm(direction)-1)<1e-8);
    map=PoincareMap_v3(Quadrupedal_Dynamics_v3(),PoincareSection_v3.apex(4), ...
        HybridSimulator_v3(),BLMarkedApexReturnPolicy_v3(struct('BLTouchdownsPerReturn',1)));
    family=EnergyFamilyResidual_v3(PeriodicOrbitResidual_v3(map,parentState),parentState,q,p,struct('Symmetry','pronk'));
    assert(isequal(family.CoordinateIndices(:),[2;3;7;8]));
    reference=[family.packState(parentState);parentRecord.critical_energy];
    normal=zeros(5,1);normal([1,3,4])=direction;
    sourceEnergy=QuadrupedEnergy_v3.evaluate(seedState,q,p);startAmplitude=direction.'*seedState([2,7,8]);
    targetAmplitude=direction.'*targetOrbit.initial_state([2,7,8]);
    assert(abs(targetAmplitude+.01)<1e-9&&startAmplitude<-.01);
    gatePaths={'Research_v3/next_round/solver/solver_validation_summary.json', ...
        'Research_v3/next_round/solver/initial_jacobian_reuse/summary.json'};
    solverGate=jsondecode(fileread(fullfile(v3,gatePaths{1})));warmGate=jsondecode(fileread(fullfile(v3,gatePaths{2})));
    assert(solverGate.all_required_gates_passed&&warmGate.passed&&warmGate.warm.initial_J_used);
    assert(strcmp(warmGate.root_sha256,RoundSHA256_v3(fullfile(v3,'Numerics_v3/RootSolver_v3.m'))));
    sourcePaths={'Dynamics_v3/ContinuousDynamics_v3.m','Dynamics_v3/GuardFunctions_v3.m', ...
        'Dynamics_v3/ResetMap_v3.m','Dynamics_v3/QuadrupedEnergy_v3.m', ...
        'Simulation_v3/EventDetector_v3.m','Simulation_v3/HybridSimulator_v3.m', ...
        'Orbit_v3/PoincareMap_v3.m','Orbit_v3/BLMarkedApexReturnPolicy_v3.m', ...
        'Orbit_v3/PeriodicOrbitResidual_v3.m','Dynamics_v3/QuadrupedPhysicalChart_v3.m', ...
        'Orbit_v3/EnergyFamilyResidual_v3.m','Orbit_v3/ConstrainedEnergyResidual_v3.m', ...
        'Orbit_v3/GaitIdentification_v3.m','Orbit_v3/PrimitiveCycleCheck_v3.m', ...
        'Orbit_v3/HybridCycleData_v3.m','Numerics_v3/PeriodicSolutionSolver_v3.m', ...
        'Numerics_v3/RootSolver_v3.m','Numerics_v3/HybridFiniteDifferenceJacobian_v3.m', ...
        'Research_v3/Drivers_v3/RunLowEnergyPKBridge_v3.m', ...
        'Research_v3/Drivers_v3/ComparePronkBridgeOrbits_v3.m', ...
        'Research_v3/next_round/theory/FreezeLowEnergyPKBridge_v3.m'};
    candidate=struct('id','imported_PK_to_low_PIP_daughter_negative_0p01', ...
        'schema_version','prospective-constrained-local-PK-bridge-v3-1','frozen_utc',char(datetime('now','TimeZone','UTC')), ...
        'parameter',p,'seed',struct('state',seedState,'mode',q,'parameter',p, ...
            'return_policy','BL-marked-apex-return','occurrence',1,'provenance',struct( ...
                'source_artifact',inputs{2},'source_point_index',2,'source_sha256',expected{2}, ...
                'source_admission','Historical saved full closure and independently audited baseline/pronk eligibility; fresh shared-service admission is still required.', ...
                'horizontal_reflection_applied',false,'roundoff_projection_error',projectionError)), ...
        'original_source_state',sourceState,'source_saved_closure',sourceClosure, ...
        'source_roundoff_projection_error',projectionError,'start_energy',sourceEnergy, ...
        'parent_energy',parentRecord.critical_energy,'parent_direction',direction, ...
        'coordinate_indices',[2;3;7;8],'observable_positions',[1;3;4], ...
        'constraint_reference',reference,'constraint_normal',normal, ...
        'start_amplitude',startAmplitude,'target_amplitude',-.01, ...
        'target_saved_observable',targetAmplitude,'amplitude_grid',linspace(startAmplitude,-.01,5), ...
        'target_orbit',targetOrbit,'target_entry',targetEntry,'target_artifact',inputs{3}, ...
        'maximum_source_admission_displacement',1e-6,'maximum_increment_state_change',5e-4, ...
        'maximum_increment_energy_change',1e-5,'maximum_wall_seconds',480, ...
        'integration',struct('RelTol',1e-11,'AbsTol',1e-13,'MaxStep',.01, ...
            'EventValueTolerance',1e-9,'EventTimeTolerance',1e-11,'SimultaneousTimeTolerance',1e-9), ...
        'replay_integration',struct('RelTol',1e-12,'AbsTol',1e-14,'MaxStep',.005, ...
            'EventValueTolerance',1e-10,'EventTimeTolerance',1e-12,'SimultaneousTimeTolerance',1e-10), ...
        'jacobian_options',struct('RelativeCandidateSteps',[1e-5,5e-6],'MaximumRelativeError',.001, ...
            'UseRichardson',true,'FailurePolicy','nan'), ...
        'acceptance',struct('FullClosureTolerance',1e-8,'EnergyTolerance',1e-8, ...
            'ConstraintTolerance',1e-8,'StanceTolerance',1e-8,'RequirePrimitive',true), ...
        'comparison_tolerances',struct('apex_state',1e-6,'energy',1e-7,'period',1e-7, ...
            'drift',1e-6,'event_phase',1e-6,'event_state',1e-6,'smooth_history',1e-4,'event_exclusion_phase',1e-6), ...
        'warm_gate_path',gatePaths{2},'solver_gate_path',gatePaths{1}, ...
        'source_hashes',hashes(sourcePaths),'input_hashes',hashes(inputs),'gate_hashes',hashes(gatePaths), ...
        'maps_executed_during_freeze',0,'global_connection_claimed',false,'rigorous_ancestry_claimed',false, ...
        'hypothesis','Four independently corrected local amplitude increments from imported PK to the frozen low-energy PIP daughter may support a local numerical continuation connection and endpoint orbit identity.');
    artifact='Research_v3/next_round/theory/low_energy_PK_bridge_candidate.mat';
    assert(~isfile(fullfile(v3,artifact)));save(fullfile(v3,artifact),'candidate','-v7');
    entry=struct('id',candidate.id,'candidate_artifact',artifact,'candidate_variable','candidate', ...
        'candidate_sha256',RoundSHA256_v3(fullfile(v3,artifact)),'maximum_wall_seconds',480);
    registration=rmfield(candidate,{'seed','target_orbit','target_entry'});registration.candidates=entry;
    registration.fresh_source_admission_counts_as_continuation_progress=false;
    registration.accepted_increments_required=4;
    RoundJSON_v3(fullfile(v3,registrationPath),registration);
    fprintf('Frozen bridge: observable %.16g -> -.01 in four increments; zero maps.\n',startAmplitude);
    function items=hashes(paths)
        items=struct('path',{},'sha256',{});
        for j=1:numel(paths),items(j)=struct('path',paths{j},'sha256',RoundSHA256_v3(fullfile(v3,paths{j})));end
    end
end
