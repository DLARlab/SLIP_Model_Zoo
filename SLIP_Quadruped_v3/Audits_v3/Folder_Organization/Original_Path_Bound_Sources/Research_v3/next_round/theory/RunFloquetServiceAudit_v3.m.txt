function reports=RunFloquetServiceAudit_v3()
%RUNFLOQUETSERVICEAUDIT_V3 Fresh full-space public Floquet-wrapper checks.
% Fine physical return differentiation is compared with the independently
% executed directional audit. The energy leaf is formed as ker(DE), with
% no numerical deletion of an eigenvalue near one.
    here=fileparts(mfilename('fullpath'));v3=fileparts(fileparts(fileparts(here)));
    paths={'Schema_v3','Dynamics_v3','Simulation_v3','Orbit_v3','Numerics_v3','Stability_v3'};
    for k=1:numel(paths),addpath(fullfile(v3,paths{k}));end
    reports=cell(2,1);targets={'imported_PK','PIP'};
    settings=struct('RelTol',1e-13,'AbsTol',1e-15,'MaxStep',.0025, ...
        'EventValueTolerance',1e-11,'EventTimeTolerance',1e-13, ...
        'SimultaneousTimeTolerance',1e-11,'ArmingTolerance',1e-12);
    for index=1:numel(targets)
        target=targets{index};
        source=load(fullfile(here,['arming_',target,'_03.mat']),'x','q','parameters');
        reference=load(fullfile(here,['directional_floquet_',target,'.mat']),'value');
        assert(reference.value.reliable_numerical_full_derivative);
        map=PoincareMap_v3(Quadrupedal_Dynamics_v3(),PoincareSection_v3.apex(4), ...
            HybridSimulator_v3(settings),BLMarkedApexReturnPolicy_v3(), ...
            struct('MaxReturnTime',10));
        options=struct('PhysicalParameter',source.parameters, ...
            'RelativeCandidateSteps',5e-5,'UseRichardson',false, ...
            'MaximumRelativeError',1e-3);
        helper=EventOrderChartJacobian_v3(options);
        analysis=FloquetAnalysis_v3(struct('Jacobian',helper));
        fprintf('Starting %s full physical Floquet wrapper at tighter ODE and FD settings\n',target);
        elapsed=tic;
        result=analysis.analyze(map,source.x,source.q,source.parameters);
        report=struct('schema_version','full-floquet-service-audit-v3-1', ...
            'matlab_version',version,'created_utc',char(datetime('now','TimeZone','UTC')), ...
            'target',target,'integration_settings',settings, ...
            'relative_fd_candidate_step',5e-5,'finest_relative_fd_step',2.5e-5, ...
            'reference_finest_relative_fd_step',5e-5, ...
            'registered_relative_matrix_tolerance',1e-3, ...
            'registered_energy_invariance_tolerance',1e-6, ...
            'physical_coordinate_dimension',result.physicalCoordinateDimension, ...
            'physical_parameters',source.parameters,'base_state',source.x, ...
            'base_mode',source.q,'matrix',result.poincareMatrix, ...
            'multipliers',result.multipliers,'spectral_radius',result.spectralRadius, ...
            'stability',result.stability,'reliable',result.reliable, ...
            'common_C1_event_order_chart',result.commonC1EventOrderChart, ...
            'raw_event_signature',result.rawEventSignature, ...
            'raw_section_signature',result.rawSectionRelativeSignature, ...
            'raw_cluster_signature',result.rawEventClusterSignature, ...
            'raw_event_signatures_match',result.rawEventSignaturesMatch, ...
            'raw_section_signatures_match',result.rawSectionSignaturesMatch, ...
            'declared_event_signature',result.eventSignature, ...
            'declared_section_signature',result.sectionRelativeSignature, ...
            'declared_cluster_signature',result.eventClusterSignature, ...
            'declared_event_signatures_match',result.eventSignaturesMatch, ...
            'declared_section_signatures_match',result.sectionSignaturesMatch, ...
            'raw_chart_observations',result.eventOrderChartObservations, ...
            'per_column_reliability',result.perColumnReliability, ...
            'finite_difference_columns',result.finiteDifferenceStepReport, ...
            'map_evaluations',result.finiteDifference.mapEvaluations, ...
            'energy_neutrality',result.energyNeutrality, ...
            'energy_leaf',result.energyLeaf, ...
            'forward_matrix',result.forwardOneSidedPoincareMatrix, ...
            'backward_matrix',result.backwardOneSidedPoincareMatrix, ...
            'warning',result.warning,'elapsed_seconds',toc(elapsed));
        report.matrix_refinement_error=norm(result.poincareMatrix ...
            -reference.value.common_coordinate_matrix,inf) ...
            /max(1,norm(result.poincareMatrix,inf));
        report.one_sided_agreement_error=norm(report.forward_matrix ...
            -report.backward_matrix,inf)/max(1,norm(report.matrix,inf));
        report.base_unshifted_state_difference=norm(result.baseMapInfo.raw_return_state ...
            -result.baseMapInfo.initial_state,inf);
        % The translation-reduced returned state gives the relevant relative
        % closure; the unshifted trajectory closure is retained separately.
        report.translation_reduced_full_closure=norm( ...
            result.perturbations(1).nextState-result.baseState,inf);
        report.actual_base_events=result.baseMapInfo.event_history;
        report.actual_base_section_chart=result.baseMapInfo.section_chart;
        report.actual_base_period=result.baseMapInfo.period;
        report.derivative_regularity_scope=result.derivativeRegularityScope;
        report.interpretation=['Numerical full physical section derivative and fixed-energy tangent restriction. ', ...
            'Conserved-energy unit remains in the full spectrum. No C2, certified existence, ', ...
            'or full asymptotic stability follows from a spectral radius near one.'];
        report.passed=report.reliable&&report.common_C1_event_order_chart ...
            &&report.physical_coordinate_dimension==12 ...
            &&report.energy_leaf.applicable&&report.energy_leaf.dimension==11 ...
            &&report.energy_leaf.reliable ...
            &&report.matrix_refinement_error<report.registered_relative_matrix_tolerance ...
            &&report.one_sided_agreement_error<report.registered_relative_matrix_tolerance ...
            &&report.energy_leaf.invariance_residual<report.registered_energy_invariance_tolerance ...
            &&report.translation_reduced_full_closure<1e-8;
        reports{index}=report;write(report);
        fprintf('%s wrapper passed%d 12D radius%.12g 11D radius%.12g refine%.3g energy-leak%.3g %d maps %.1fs\n', ...
            target,report.passed,report.spectral_radius,report.energy_leaf.spectral_radius, ...
            report.matrix_refinement_error,report.energy_leaf.invariance_residual, ...
            report.map_evaluations,report.elapsed_seconds);
        assert(report.passed,'The full physical Floquet wrapper gate failed.');
    end
    function write(report)
        path=fullfile(here,['floquet_service_',report.target,'.mat']);
        save(path,'report','-v7');
        document=report;
        document.multipliers=complexRecord(report.multipliers);
        document.energy_leaf.multipliers=complexRecord(report.energy_leaf.multipliers);
        fid=fopen(fullfile(here,['floquet_service_',report.target,'.json']),'w');assert(fid>=0);
        cleanup=onCleanup(@()fclose(fid)); %#ok<NASGU>
        fprintf(fid,'%s\n',jsonencode(document,'PrettyPrint',true));
    end
    function value=complexRecord(numbers)
        value=struct('real',real(numbers),'imag',imag(numbers));
    end
end
