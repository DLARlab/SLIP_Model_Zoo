function audit=AuditEvidencePromotions_v3()
%AUDITEVIDENCEPROMOTIONS_V3 Independently inspect executed orbit/graph evidence.
% This reads a graph snapshot and frozen orbit artifacts; it never edits the
% master checkpoint, task queue, report or graph. No new solve is represented.
    here=fileparts(mfilename('fullpath'));v3=fileparts(fileparts(fileparts(here)));
    for folder={'Schema_v3','Dynamics_v3','Simulation_v3','Orbit_v3','Numerics_v3','Research_v3/Drivers_v3'}
        addpath(fullfile(v3,folder{1}));
    end
    graphPath='Research_v3/next_round/graph/index_full.json';
    graphText=fileread(fullfile(v3,graphPath));graph=jsondecode(graphText);
    selected=graph.nodes(arrayfun(@(node)strcmp(node.type,'recovered_orbit') || ...
        strcmp(node.id,'PIP_candidate_1_daughter_1'),graph.nodes));
    records=struct([]);
    for k=1:numel(selected)
        node=selected(k);saved=load(fullfile(v3,node.evidence));
        if isfield(saved,'solution')
            orbit=saved.solution;reported=saved.observation;reportedLabel=reported.gait;
            declaredAccepted=reported.accepted;recordedClosure=reported.final_closure;
        else
            orbit=saved.orbit;reported=saved.entry;reportedLabel=char(reported.gait);
            declaredAccepted=reported.accepted;recordedClosure=reported.full_closure;
        end
        assert(~isempty(orbit)&&declaredAccepted,'Positive node lacks an accepted physical orbit.');
        gait=GaitIdentification_v3(orbit);data=HybridCycleData_v3.unpack(orbit);
        actualIds=[data.history.guard_id];counts=arrayfun(@(id)sum(actualIds==id),1:8);
        record=struct('id',node.id,'artifact',node.evidence, ...
            'artifact_sha256',RoundSHA256_v3(fullfile(v3,node.evidence)), ...
            'graph_type',node.type,'graph_scope',node.scope, ...
            'reported_label',reportedLabel,'recomputed_label',char(gait.label), ...
            'classification_status',char(gait.status), ...
            'primitive_status',char(gait.primitive.status), ...
            'checked_cover_bound',gait.primitive.checked_cover_bound, ...
            'global_primitivity_proved',gait.primitive.global_primitivity_proved, ...
            'physical_flight_count',gait.flight_count, ...
            'raw_positive_flight_count',gait.raw_positive_flight_count, ...
            'flight_count_sensitivity',gait.flight_count_sensitivity, ...
            'actual_event_ids',actualIds,'contact_counts_TD_then_LO',counts, ...
            'all_legs_one_TD_and_LO',all(counts==1), ...
            'maximum_contact_sync_error',max(gait.contact_sync_errors,[],'all'), ...
            'maximum_motion_sync_error',max(gait.motion_sync_errors,[],'all'), ...
            'primitive_drift',gait.primitive.primitive_drift, ...
            'energy',QuadrupedEnergy_v3.evaluate(orbit.initial_state,orbit.initial_mode,orbit.parameter), ...
            'period',orbit.period,'recorded_full_closure',recordedClosure, ...
            'trajectory_closure',gait.primitive.closure_error, ...
            'label_matches',strcmp(reportedLabel,char(gait.label)));
        if isempty(records),records=record;else,records(end+1)=record;end %#ok<AGROW>
    end
    nonUnresolved=graph.edges(~arrayfun(@(edge)strcmp(edge.type,'unresolved_candidate'),graph.edges));
    audit=struct('schema_version','independent-evidence-promotion-audit-v3-1', ...
        'created_utc',char(datetime('now','TimeZone','UTC')),'matlab_version',version, ...
        'graph_snapshot_artifact',graphPath,'graph_snapshot_sha256',RoundSHA256_v3(fullfile(v3,graphPath)), ...
        'graph_network_achieved',graph.network_achieved, ...
        'graph_global_completeness_claimed',graph.global_completeness_claimed, ...
        'positive_orbit_records_checked',numel(records),'records',records, ...
        'non_unresolved_edges',nonUnresolved, ...
        'passed',all([records.label_matches])&&all([records.all_legs_one_TD_and_LO])&& ...
            all([records.physical_flight_count]==1)&&~graph.network_achieved&&~graph.global_completeness_claimed, ...
        'scope',['Saved complete hybrid trajectories were reclassified independently. ', ...
            'This is an evidence-promotion review, not a new independent dynamics replay or a global primitive-period proof. ', ...
            'Recovered record counts may duplicate orbit states; descriptive labels supply no genealogy or full stability.']);
    RoundSave_v3(fullfile(here,'independent_evidence_promotion_audit.mat'),struct('audit',audit));
    RoundJSON_v3(fullfile(here,'independent_evidence_promotion_audit.json'),audit);
    assert(audit.passed,'Positive evidence failed independent gait/claim review.');
    fprintf('Independent saved-trace promotion audit passed: %d PK orbit records, one flight and one TD/LO per leg each.\n',numel(records));
end
