function audit=AuditOpposedSpreadPhaseEquivalence_v3()
%AUDITOPPOSEDSPREADPHASEEQUIVALENCE_V3 Read saved BL1 returns; no simulation.
% Opposite signed predictors at fixed n and |A| are marked phases of the
% same primitive BL2 orbit. No leg relabeling or new closure is performed.
    here=fileparts(mfilename('fullpath'));v3=fileparts(fileparts(fileparts(here)));
    for folder={'Schema_v3','Dynamics_v3','Simulation_v3','Orbit_v3','Numerics_v3','Research_v3/Drivers_v3'}
        addpath(fullfile(v3,folder{1}));
    end
    audit=struct('schema_version','saved-opposed-spread-phase-equivalence-v3-1', ...
        'created_utc',char(datetime('now','TimeZone','UTC')),'matlab_version',version, ...
        'exact_relation','P_BL1(x_n(A))=x_n(-A), P_BL2(x_n(A))=x_n(A)', ...
        'equivalence','Same labeled physical cycle after one half-period time shift, without leg relabeling', ...
        'new_simulations',0,'new_map_evaluations',0,'new_closure_certification',false, ...
        'case_records',struct([]),'signed_predictor_records',4,'sampled_orbit_classes_modulo_time_phase',2, ...
        'distinct_global_gait_families_established',false,'passed',false);
    for n=[0,1]
        positivePath=sprintf('Research_v3/next_round/theory/opposed_spread_n%d_positive_A1e-3_actual_trace.mat',n);
        negativePath=sprintf('Research_v3/next_round/theory/opposed_spread_n%d_negative_A1e-3_actual_trace.mat',n);
        positiveHash=RoundSHA256_v3(fullfile(v3,positivePath));negativeHash=RoundSHA256_v3(fullfile(v3,negativePath));
        positive=load(fullfile(v3,positivePath),'predictor','single','info1','info2');
        negative=load(fullfile(v3,negativePath),'predictor','single','info1','info2');
        assert(strcmp(positiveHash,RoundSHA256_v3(fullfile(v3,positivePath))) ...
            &&strcmp(negativeHash,RoundSHA256_v3(fullfile(v3,negativePath))), ...
            'An immutable actual trace changed while being read.');
        item=struct('resonance_index',n,'absolute_TD_angle',abs(positive.predictor.touchdown_angle), ...
            'positive_trace_artifact',positivePath,'positive_trace_sha256',positiveHash, ...
            'negative_trace_artifact',negativePath,'negative_trace_sha256',negativeHash, ...
            'positive_BL1_to_negative_predictor_residual',norm(positive.single-negative.predictor.state,inf), ...
            'negative_BL1_to_positive_predictor_residual',norm(negative.single-positive.predictor.state,inf), ...
            'counterpart_modes_agree',isequal(positive.info1.final_mode,negative.predictor.mode) ...
                &&isequal(negative.info1.final_mode,positive.predictor.mode), ...
            'same_energy',positive.predictor.energy==negative.predictor.energy, ...
            'positive_BL1_duration',positive.info1.period,'negative_BL1_duration',negative.info1.period, ...
            'half_duration_difference',abs(positive.info1.period-negative.info1.period), ...
            'positive_full_BL2_duration',positive.info2.period,'negative_full_BL2_duration',negative.info2.period, ...
            'full_duration_difference',abs(positive.info2.period-negative.info2.period), ...
            'BL1_half_of_BL2_time_error',max(abs([positive.info1.period-positive.info2.period/2, ...
                negative.info1.period-negative.info2.period/2])), ...
            'time_phase_shift_fraction',.5,'leg_permutation_applied',false,'new_orbit_discovered',false);
        item.passed=max([item.positive_BL1_to_negative_predictor_residual,item.negative_BL1_to_positive_predictor_residual, ...
            item.half_duration_difference,item.full_duration_difference,item.BL1_half_of_BL2_time_error])<1e-8 ...
            &&item.counterpart_modes_agree&&item.same_energy;
        if isempty(audit.case_records),audit.case_records=item;else,audit.case_records(end+1)=item;end %#ok<AGROW>
        fprintf('Opposed spread n%d: saved +BL1→- %.3g, -BL1→+ %.3g, half-period error %.3g\n', ...
            n,item.positive_BL1_to_negative_predictor_residual,item.negative_BL1_to_positive_predictor_residual,item.BL1_half_of_BL2_time_error);
    end
    audit.passed=all([audit.case_records.passed]);
    save(fullfile(here,'opposed_spread_phase_equivalence.mat'),'audit','-v7');
    RoundJSON_v3(fullfile(here,'opposed_spread_phase_equivalence.json'),audit);
    assert(audit.passed,'Saved signed BL1 states did not pass the phase-equivalence comparison.');
end
