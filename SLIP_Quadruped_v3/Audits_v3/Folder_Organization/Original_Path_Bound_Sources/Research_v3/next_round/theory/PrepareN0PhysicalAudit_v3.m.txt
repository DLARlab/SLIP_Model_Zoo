function comparison=PrepareN0PhysicalAudit_v3()
%PREPAREN0PHYSICALAUDIT_V3 Independent immutable daughter/source comparison.
% Equal descriptive PK labels do not identify an orbit or its ancestry.
    here=fileparts(mfilename('fullpath'));v3=fileparts(fileparts(fileparts(here)));
    for folder={'Schema_v3','Dynamics_v3','Simulation_v3','Orbit_v3','Adapters_v3'}
        addpath(fullfile(v3,folder{1}));
    end
    daughterPath='Research_v3/next_round/tasks/full/PIP_candidate_1/daughter_-0.0001_attempt_1.mat';
    daughter=load(fullfile(v3,daughterPath),'orbit','seed','entry');assert(~isempty(daughter.orbit)&&daughter.entry.accepted);
    n0= daughter.orbit;parameters=n0.parameter;q=n0.initial_mode;
    record=jsondecode(fileread(fullfile(v3,'Research_v3/next_round/tasks/full/PIP_candidate_1/phase_checkpoint.json')));
    criticalEnergy=record.critical_energy;x=zeros(14,1);x(3)=criticalEnergy;
    parentProvenance=struct('source','Analytic vertical PIP lift at the saved n0 critical energy', ...
        'critical_energy',criticalEnergy,'parameter_source',daughterPath, ...
        'daughter_artifact_sha256',fileHash(fullfile(v3,daughterPath)));
    save(fullfile(here,'arming_n0_parent_03.mat'),'x','q','parameters','parentProvenance','-v7');
    inventory=load(fullfile(v3,'Research_v3/next_round/source_inventory.mat'),'inventory');
    records=inventory.inventory.records;
    index=find(arrayfun(@(r)strcmp(r.source.fixture_path, ...
        'P2_All_Common_Quadrupedal_Gaits/SourceFixtures_v3/PK_20_2.mat'),records),1);
    source=records(index);variable=source.variables(1);[~,column]=min(variable.energy);
    turnState=variable.mapped_initial_state(:,column);turnMode=variable.mapped_initial_mode(:,column);
    turnParameter=variable.mapped_physical_parameters(:,column);
    assert(isequal(parameters,turnParameter),'The physical models differ.');
    settings=struct('RelTol',1e-12,'AbsTol',1e-14,'MaxStep',.005, ...
        'EventValueTolerance',1e-10,'EventTimeTolerance',1e-12, ...
        'SimultaneousTimeTolerance',1e-10,'ArmingTolerance',1e-11);
    map=PoincareMap_v3(Quadrupedal_Dynamics_v3(),PoincareSection_v3.apex(4), ...
        HybridSimulator_v3(settings),BLMarkedApexReturnPolicy_v3(),struct('MaxReturnTime',10));
    [daughterReturn,daughterInfo]=map.evaluate(n0.initial_state,n0.initial_mode,parameters);
    [turnReturn,turnInfo]=map.evaluate(turnState,turnMode,turnParameter);
    [parentReturn,parentInfo]=map.evaluate(x,q,parameters);
    daughterEnergy=QuadrupedEnergy_v3.evaluate(n0.initial_state,n0.initial_mode,parameters);
    turnEnergy=QuadrupedEnergy_v3.evaluate(turnState,turnMode,turnParameter);
    comparison=struct('schema_version','n0-daughter-imported-PK-turn-comparison-v3-1', ...
        'matlab_version',version,'created_utc',char(datetime('now','TimeZone','UTC')), ...
        'settings',settings,'same_physical_parameters',true,'physical_parameters',parameters, ...
        'n0_critical_parent_energy',criticalEnergy,'n0_parent_state',x, ...
        'n0_parent_full_closure',norm(parentReturn-x,inf),'n0_parent_period',parentInfo.period, ...
        'daughter_source_artifact',daughterPath,'daughter_source_sha256',parentProvenance.daughter_artifact_sha256, ...
        'daughter_initial_state',n0.initial_state,'daughter_energy',daughterEnergy, ...
        'daughter_full_closure',norm(daughterReturn-n0.initial_state,inf), ...
        'daughter_period',daughterInfo.period, ...
        'daughter_mean_speed',(daughterInfo.raw_return_state(1)-daughterInfo.initial_state(1))/daughterInfo.period, ...
        'daughter_reported_label',char(daughter.entry.gait), ...
        'daughter_reported_primitive',daughter.entry.classification.primitive.status, ...
        'daughter_reported_flight_count',daughter.entry.classification.flight_count, ...
        'daughter_actual_event_ids',contactIds(daughterInfo), ...
        'turn_source_fixture',source.source.fixture_path,'turn_source_sha256',source.fixture_sha256, ...
        'turn_source_variable',variable.variable,'turn_source_column',column, ...
        'turn_initial_state',turnState,'turn_energy',turnEnergy, ...
        'turn_full_closure',norm(turnReturn-turnState,inf),'turn_period',turnInfo.period, ...
        'turn_mean_speed',(turnInfo.raw_return_state(1)-turnInfo.initial_state(1))/turnInfo.period, ...
        'turn_actual_event_ids',contactIds(turnInfo), ...
        'energy_gap',turnEnergy-daughterEnergy, ...
        'apex_state_distance',norm(turnState-n0.initial_state,inf), ...
        'interpretation',['Distinct energies prove these are not the same conservative orbit under rephasing/translation. ', ...
            'Both may have a descriptive synchronized PK label; that does not establish or exclude a connecting branch. ', ...
            'No attachment to the imported PK turn or common ancestry is claimed.']);
    save(fullfile(here,'n0_source_comparison.mat'),'comparison','-v7');
    fid=fopen(fullfile(here,'n0_source_comparison.json'),'w');assert(fid>=0);
    cleanup=onCleanup(@()fclose(fid)); %#ok<NASGU>
    fprintf(fid,'%s\n',jsonencode(comparison,'PrettyPrint',true));
    fprintf('n0 daughter E%.12g versus imported PK turn col%d E%.12g: gap%.6g, closure%.3g/%.3g\n', ...
        daughterEnergy,column,turnEnergy,comparison.energy_gap,comparison.daughter_full_closure,comparison.turn_full_closure);
    function ids=contactIds(info)
        events=info.event_history;events=events(~[events.is_stop]);ids=[events.guard_id];
    end
    function digest=fileHash(path)
        fid=fopen(path,'rb');assert(fid>=0);cleanup=onCleanup(@()fclose(fid)); %#ok<NASGU>
        engine=java.security.MessageDigest.getInstance('SHA-256');
        while ~feof(fid),engine.update(fread(fid,1024*1024,'*uint8'));end
        digest=lower(reshape(dec2hex(typecast(engine.digest(),'uint8'),2).',1,[]));
    end
end
