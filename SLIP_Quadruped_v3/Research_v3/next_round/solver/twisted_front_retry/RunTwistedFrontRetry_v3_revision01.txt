function result=RunTwistedFrontRetry_v3(record,cfg,outputDirectory,maxWallSeconds,task)
%RUNTWISTEDFRONTRETRY_V3 One registered corrector, actual physical BL2 replay.
% This native driver writes only its own result directory. The master owns
% task dispatch, total wall budget, queue state and any subsequent retry.
    v3=fileparts(fileparts(fileparts(mfilename('fullpath'))));clock=tic;
    if nargin<5,task=struct('repair_revision',0);end
    outputDirectory=V3OutputPath_v3(outputDirectory);if ~isfolder(outputDirectory),mkdir(outputDirectory);end
    method=jsondecode(fileread(fullfile(v3,record.method_registration)));
    assert(strcmp(method.method,'front-twisted-BL1-common-C1-Broyden-retry')&&~method.initial_J_reuse);
    for items={method.source_hashes,method.canonical_source_hashes,method.original_physical_engine_hashes,method.physical_model_hashes}
        hashes=items{1};
        for k=1:numel(hashes)
            assert(strcmp(hashes(k).sha256,RoundSHA256_v3(fullfile(v3,hashes(k).path))), ...
                'RunTwistedFrontRetry_v3:SourceEpoch','A registered source changed: %s.',hashes(k).path);
        end
    end
    gate=jsondecode(fileread(fullfile(v3,method.canonical_gate)));
    assert(gate.all_required_gates_passed&&all([gate.cases.passed])&&numel(gate.cases)>=7);
    assert(strcmp(method.canonical_gate_sha256,RoundSHA256_v3(fullfile(v3,method.canonical_gate))), ...
        'RunTwistedFrontRetry_v3:GateEpoch','Canonical gate changed after prospective registration.');
    metadata=jsondecode(fileread(fullfile(v3,method.metadata_gate)));
    assert(metadata.passed&&metadata.missing_event_history_rejected&&metadata.map_info_unchanged);
    assert(strcmp(method.metadata_gate_sha256,RoundSHA256_v3(fullfile(v3,method.metadata_gate))) ...
        &&strcmp(record.candidate_sha256,RoundSHA256_v3(fullfile(v3,record.candidate_artifact))));
    loaded=load(fullfile(v3,record.candidate_artifact),record.candidate_variable);candidate=loaded.(record.candidate_variable);
    p=candidate.physical_parameter(:);q=candidate.mode(:);w=candidate.coordinates(:);
    assert(numel(w)==13&&isempty(candidate.initial_jacobian)&&candidate.seed.occurrence==2);
    one=family(1);two=family(2);
    assert(isequal(one.CoordinateIndices(:),candidate.coordinate_indices(:)));
    residual=TwistedReturnFamilyResidual_v3(one,two,candidate.constraint,struct('CommonC1BroydenMetadata',true));
    seed=candidate.seed;seed.state=residual.fullState(w);seed.occurrence=2;
    assert(norm(seed.state-candidate.seed.state,inf)<1e-12,'Saved best candidate and physical chart disagree.');
    seed.provenance.method_registration=record.method_registration;seed.provenance.repair_revision=task.repair_revision;
    derivative=EventOrderChartJacobian_v3(struct('PhysicalParameter',p,'BLTouchdownsPerReturn',1, ...
        'RelativeCandidateSteps',method.fd_relative_candidate_steps(:).', ...
        'MaximumRelativeError',method.fd_maximum_relative_error,'UseRichardson',method.fd_use_richardson));
    limit=min(maxWallSeconds,method.maximum_correction_wall_seconds);
    options=struct('Residual',residual,'Coordinates',w,'AugmentedParameter',[p;w(end)], ...
        'MaxWallSeconds',max(.001,limit-toc(clock)-5),'MaxReturnTime',cfg.domain.primitive_period_max, ...
        'MaxCycleEvents',cfg.domain.event_count_max,'ReplayIntegration',method.replay_integration, ...
        'Acceptance',method.acceptance,'SolverOptions',struct('Algorithm','newton', ...
        'MaxIterations',min(cfg.budgets.root_max_iterations,method.maximum_newton_iterations), ...
        'MaxFunctionEvaluations',min(cfg.budgets.root_max_function_evaluations,method.maximum_function_evaluations), ...
        'Jacobian',derivative,'UseBroyden',true,'JacobianRefreshInterval',method.Jacobian_refresh_accepted_steps, ...
        'FunctionTolerance',method.function_tolerance, ...
        'ResidualAcceptanceTolerance',method.residual_acceptance_tolerance, ...
        'ReuseJacobian',false,'InitialJacobian',[]));
    RoundSave_v3(fullfile(outputDirectory,'front_retry_input.mat'),struct('candidate',candidate,'seed',seed,'options',options,'method',method));
    [orbit,report]=PeriodicSolutionSolver_v3(seed,options);
    state='retryable_failure';scientific='front_twisted_candidate_unresolved';
    if report.accepted
        state='accepted';scientific='primitive_front_twisted_BL2_orbit_admitted_ancestry_unresolved';
        assert(orbit.section_chart.BL_touchdowns_per_return==2&&report.primitive.status=="primitive_within_checked_bound");
    elseif toc(clock)>=limit-15
        state='partial_checkpoint';
    end
    trial=struct('accepted',report.accepted,'state',state,'coordinates',member(report,'final_coordinates',w), ...
        'best_candidate',report.final_candidate,'initial_residual',report.initial_scaled_norm, ...
        'final_residual',report.final_scaled_norm,'full_BL2_closure',report.full_closure, ...
        'constraint_residual',report.constraint_residual,'primary_failure',report.primary_failure, ...
        'finalization_failure',report.finalization_failure,'replay_failure',report.replay_failure, ...
        'elapsed_seconds',report.elapsed_seconds,'initial_J_reused',false,'accepted_Newton_steps',0, ...
        'artifact','front_retry_report.mat','classification',report.classification,'primitive',report.primitive);
    if isfield(report.solver,'acceptedNewtonIterations'),trial.accepted_Newton_steps=report.solver.acceptedNewtonIterations;end
    result=struct('schema_version','native-front-twisted-retry-result-v3-1','state',state,'scientific_status',scientific, ...
        'candidate_id',record.id,'candidate_artifact',record.candidate_artifact,'correction_method',method.method, ...
        'front_retry',trial,'new_points',double(report.accepted),'accepted_record_count',double(report.accepted), ...
        'function_evaluations',member(report.solver,'functionEvaluationCount',0), ...
        'solver_map_evaluations',member(report.solver,'mapEvaluationCount',0), ...
        'ancestry_claimed',false,'target_gait_claimed',false,'unrestricted_C2_claimed',false, ...
        'message','Actual BL1 twisted corrector with verified compatibility metadata; actual independent physical BL2 acceptance only.');
    RoundSave_v3(fullfile(outputDirectory,'front_retry_report.mat'),struct('orbit',orbit,'report',report,'result',result,'options',options));
    RoundJSON_v3(fullfile(outputDirectory,'front_retry_result.json'),result);
    function familyObject=family(count)
        policy=BLMarkedApexReturnPolicy_v3(struct('BLTouchdownsPerReturn',count));
        map=PoincareMap_v3(Quadrupedal_Dynamics_v3(),PoincareSection_v3.apex(4), ...
            HybridSimulator_v3(method.integration),policy,struct('MaxReturnTime',cfg.domain.primitive_period_max, ...
            'MaxCycleEvents',cfg.domain.event_count_max));
        familyObject=EnergyFamilyResidual_v3(PeriodicOrbitResidual_v3(map,candidate.reference_state), ...
            candidate.reference_state,q,p,struct('ClosureTolerance',method.acceptance.FullClosureTolerance));
    end
    function value=member(object,name,fallback)
        value=fallback;if isstruct(object)&&isfield(object,name),value=object.(name);end
    end
end
